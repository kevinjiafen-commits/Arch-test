-- ============================================================
-- Utility Functions
-- ============================================================
function Lib:GetDarkerColor(color)
    local h, s, v = Color3.toHSV(color)
    return Color3.fromHSV(h, s, v / 1.5)
end

function Lib:MapValue(value, inMin, inMax, outMin, outMax)
    return ((value - inMin) / (inMax - inMin)) * (outMax - outMin) + outMin
end

function Lib:GetTextBounds(text, font, size, width)
    local w = width or 1920
    local x, y = TextService:GetTextBoundsAsync(
        TextService:GetTextBoundsParams(text, size, font, Vector2.new(w, 1000))
    )
    return x, y
end

local SaveManager = {}
SaveManager.__index = SaveManager

function SaveManager:new(folder)
    local obj = setmetatable({ Folder = folder or "Configs" }, self)
    obj:BuildFolderTree()
    obj.Parsers = {
        Toggle = {
            Save = function(id, ctrl) return { type = "Toggle", idx = id, value = ctrl.Value } end,
            Load = function(id, data)
                local ctrl = Lib.Controls[id]
                if ctrl then ctrl:SetValue(data.value) end
            end
        },
        Slider = {
            Save = function(id, ctrl) return { type = "Slider", idx = id, value = ctrl.Value } end,
            Load = function(id, data)
                local ctrl = Lib.Controls[id]
                if ctrl then ctrl:SetValue(data.value) end
            end
        },
        Dropdown = {
            Save = function(id, ctrl) return { type = "Dropdown", idx = id, value = ctrl.Value, multi = ctrl.Multi } end,
            Load = function(id, data)
                local ctrl = Lib.Controls[id]
                if ctrl then ctrl:SetValue(data.value) end
            end
        },
        ColorPicker = {
            Save = function(id, ctrl) return { type = "ColorPicker", idx = id, value = ctrl.Value:ToHex(), transparency = ctrl.Transparency } end,
            Load = function(id, data)
                local ctrl = Lib.Controls[id]
                if ctrl then ctrl:SetValueRGB(Color3.fromHex(data.value), data.transparency) end
            end
        },
        KeybindPicker = {
            Save = function(id, ctrl) return { type = "KeybindPicker", idx = id, mode = ctrl.Mode, key = ctrl.Value } end,
            Load = function(id, data)
                local ctrl = Lib.Controls[id]
                if ctrl then ctrl:SetValue({ data.key, data.mode }) end
            end
        }
    }
    return obj
end

function SaveManager:BuildFolderTree()
    for _, sub in ipairs({ "", "/settings", "/themes" }) do
        local path = self.Folder .. sub
        if not isfolder(path) then makefolder(path) end
    end
end

function SaveManager:Save(fileName, silent)
    if not fileName then return false, "No config file selected" end
    local data = { objects = {} }
    for id, ctrl in pairs(Lib.Controls) do
        local parser = self.Parsers[ctrl.Type]
        if parser then
            table.insert(data.objects, parser.Save(id, ctrl))
        end
    end
    local json = HttpService:JSONEncode(data)
    writefile(self.Folder .. "/settings/" .. fileName .. ".json", json)
    if not silent then Lib:Notify("Saved config: " .. fileName) end
    return true
end

function SaveManager:Load(fileName)
    if not fileName then return false, "No config file selected" end
    local path = self.Folder .. "/settings/" .. fileName .. ".json"
    if not isfile(path) then return false, "Config not found" end
    local data = HttpService:JSONDecode(readfile(path))
    for _, obj in ipairs(data.objects) do
        local parser = self.Parsers[obj.type]
        if parser then
            parser.Load(obj.idx, obj)
        end
    end
    Lib:Notify("Loaded config: " .. fileName)
    return true
end

function SaveManager:RefreshConfigList()
    local list = {}
    local path = self.Folder .. "/settings/"
    if isfolder(path) then
        for _, file in ipairs(listfiles(path)) do
            if file:match("%.json$") then
                local name = file:match("([^/\\]+)%.json$")
                if name then table.insert(list, name) end
            end
        end
    end
    return list
end

function SaveManager:Delete(fileName)
    if not fileName then return false end
    local path = self.Folder .. "/settings/" .. fileName .. ".json"
    if isfile(path) then
        delfile(path)
        Lib:Notify("Deleted config: " .. fileName)
        return true
    end
    return false
end

-- ============================================================
-- Global Keybind Handling
-- ============================================================
function Lib:BindGlobalKeybinds()
    self:GiveSignal(UserInputService.InputBegan:Connect(function(input, gameProcessed)
        if gameProcessed then return end
        for _, ctrl in pairs(self.Controls) do
            if ctrl.Type == "KeybindPicker" then
                local key = ctrl.Value
                local mode = ctrl.Mode
                if key and key ~= "..." then
                    if input.KeyCode.Name == key and mode ~= "Always" then
                        if mode == "Toggle" then
                            ctrl.Toggled = not ctrl.Toggled
                        elseif mode == "Hold" then
                            ctrl.Toggled = true
                        end
                        ctrl:Update()
                    end
                end
            end
        end
    end))

    self:GiveSignal(UserInputService.InputEnded:Connect(function(input)
        for _, ctrl in pairs(self.Controls) do
            if ctrl.Type == "KeybindPicker" and ctrl.Mode == "Hold" then
                if input.KeyCode.Name == ctrl.Value then
                    ctrl.Toggled = false
                    ctrl:Update()
                end
            end
        end
    end))
end

                t.Button.TextColor3 = Lib.FontColor

            self.Content.Visible = true
            self.Button.BackgroundColor3 = Lib.AccentColorDark
            self.Button.TextColor3 = Lib.FontColor

            local totalHeight = 0
            for _, child in ipairs(content:GetChildren()) do
                if child:IsA("Frame") and child.Visible then
                    totalHeight = totalHeight + child.Size.Y.Offset
                end
            end
            content.Size = UDim2.new(1, 0, 0, totalHeight + 10)
            window:Resize()
        end

        table.insert(window.Tabs, tab)
        if default then tab:Show() end
        return tab
    end

    function window:Resize()
        local total = 0
        for _, tab in pairs(window.Tabs) do
            if tab.Visible then
                total = tab.Content.Size.Y.Offset + (UserInputService.TouchEnabled and 84 or 78)
                break
            end
        end
        local viewport = workspace.CurrentCamera and workspace.CurrentCamera.ViewportSize or Vector2.new(800, 600)
        local mobile = UserInputService.TouchEnabled
        local minHeight = mobile and math.min(520, math.max(360, viewport.Y - 90)) or 450
        local maxHeight = mobile and math.min(700, math.max(360, viewport.Y - 24)) or 650
        local desired = math.max(total + 8, mobile and 360 or 360)
        desired = math.min(desired, maxHeight)
        desired = math.max(desired, minHeight)
        local width = mainFrame.Size.X.Offset
        mainFrame.Size = UDim2.fromOffset(width, desired)
        mainFrame.Position = UDim2.new(0.5, 0, 0.5, 0)
    end

    self.OpenedWindows = self.OpenedWindows or {}
    table.insert(self.OpenedWindows, window)
    window.Opened = true

    return window
end

-- Toggle
function Lib:CreateToggle(parent, config)
    config = config or {}
    local ctrl = {
        Type = "Toggle",
        Value = config.Default or false,
        Callback = config.Callback or function() end,
        ChangedFuncs = {},
        Risky = config.Risky or false,
        Idx = config.Idx or ("toggle_" .. tostring(#self.Controls + 1)),
    }
    self.Controls[ctrl.Idx] = ctrl

    local frame = Instance.new("Frame")
    frame.Parent = parent
    frame.BackgroundColor3 = self.MainColor
    frame.BorderColor3 = self.OutlineColor
    frame.BorderMode = Enum.BorderMode.Middle
    frame.Size = UDim2.new(1, -2, 0, 28)
    frame.ZIndex = 1
    frame.Active = true

    local label = Instance.new("TextLabel")
    label.Parent = frame
    label.BackgroundTransparency = 1
    label.Text = config.Text or "Toggle"
    label.Font = self.Font
    label.TextColor3 = self.FontColor
    label.TextSize = 14
    label.Size = UDim2.new(1, -30, 1, 0)
    label.Position = UDim2.new(0, 5, 0, 0)
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.ZIndex = 2

    local toggleBtn = Instance.new("TextButton")
    toggleBtn.Parent = frame
    toggleBtn.BackgroundColor3 = self.MainColor
    toggleBtn.BorderColor3 = self.OutlineColor
    toggleBtn.BorderMode = Enum.BorderMode.Middle
    toggleBtn.Size = UDim2.new(0, 22, 1, -4)
    toggleBtn.Position = UDim2.new(1, -26, 0, 2)
    toggleBtn.ZIndex = 2
    toggleBtn.Text = ""
    toggleBtn.Active = true

    local check = Instance.new("Frame")
    check.Parent = toggleBtn
    check.BackgroundColor3 = self.AccentColor
    check.Size = UDim2.new(0.6, 0, 0.6, 0)
    check.Position = UDim2.new(0.2, 0, 0.2, 0)
    check.ZIndex = 3
    check.Visible = ctrl.Value

    function ctrl:SetValue(value)
        value = not not value
        self.Value = value
        check.Visible = value
        frame.BackgroundColor3 = value and self.AccentColor or self.MainColor
        frame.BorderColor3 = value and self.AccentColorDark or self.OutlineColor
        for _, func in ipairs(self.ChangedFuncs) do
            func(value)
        end
        self.Callback(value)
        self:UpdateDependencies()
        Lib.SaveManager:Save(nil, true)
    end

    function ctrl:OnChanged(func)
        table.insert(self.ChangedFuncs, func)
        func(self.Value)
    end

    function ctrl:UpdateDependencies()
        for _, box in pairs(Lib.DependencyBoxes) do
            box:Update()
        end
    end

    function ctrl:Remove()
        frame:Destroy()
        self.ChangedFuncs = nil
        Lib.Controls[ctrl.Idx] = nil
    end

    toggleBtn.MouseButton1Click:Connect(function()
        ctrl:SetValue(not ctrl.Value)
    end)

    if self.FlagCopying then
        local menu = self:AddContextMenu(toggleBtn, frame)
        menu:AddOption("Copy Flag", function()
            setclipboard(ctrl.Idx)
            self:Notify("Copied flag to clipboard!")
            menu:Hide()
        end)
    end

    self:AddToRegistry(frame, { BackgroundColor3 = "MainColor", BorderColor3 = "OutlineColor" })
    self:AddToRegistry(check, { BackgroundColor3 = "AccentColor" })

    if ctrl.Risky then
        label.TextColor3 = self.RiskColor
        self:AddToRegistry(label, { TextColor3 = "RiskColor" })
    end

    return ctrl
end

        if self.Compact then
            valueLabel.Text = self.Value .. self.Suffix
        elseif self.HideMax then
            valueLabel.Text = string.format("%s", self.Value .. self.Suffix)
        else
            valueLabel.Text = string.format("%s/%s", self.Value .. self.Suffix, self.Max .. self.Suffix)
        end
    end

    function ctrl:SetValue(value)
        local val = tonumber(value)
        if not val then return end
        val = math.clamp(val, self.Min, self.Max)
        if self.Increment then
            val = math.round(val / self.Increment) * self.Increment
        end
        if self.Rounding >= 0 then
            val = tonumber(string.format("%." .. self.Rounding .. "f", val))
        end
        self.Value = val
        self:UpdateDisplay()
        for _, func in ipairs(self.ChangedFuncs) do
            func(val)
        end
        self.Callback(val)
        Lib.SaveManager:Save(nil, true)
    end

    function ctrl:OnChanged(func)
        table.insert(self.ChangedFuncs, func)
        func(self.Value)
    end

    function ctrl:Remove()
        frame:Destroy()
        self.ChangedFuncs = nil
        Lib.Controls[ctrl.Idx] = nil
    end

    local dragging = false
    local startX, startOffset

    dragArea.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or
           input.UserInputType == Enum.UserInputType.Touch then
            if self.IsDragging then return end
            dragging = true
            local pos = input.UserInputType == Enum.UserInputType.MouseButton1 and
                UserInputService:GetMouseLocation() or input.Position
            startX = pos.X
            startOffset = fill.Size.X.Offset
        end
    end)

    dragArea.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement or
           input.UserInputType == Enum.UserInputType.Touch then
            local pos = input.UserInputType == Enum.UserInputType.MouseMovement and
                UserInputService:GetMouseLocation() or input.Position
            local delta = pos.X - startX
            local maxWidth = frame.AbsoluteSize.X - 4
            local newOffset = math.clamp(startOffset + delta, 0, maxWidth)
            local val = self:MapValue(newOffset, 0, maxWidth, self.Min, self.Max)
            self:SetValue(val)
        end
    end)

    dragArea.InputEnded:Connect(function(input)
        if dragging then
            dragging = false
            Lib.SaveManager:Save(nil, true)
        end
    end)

    if self.FlagCopying then
        local menu = self:AddContextMenu(frame, frame)
        menu:AddOption("Copy Flag", function()
            setclipboard(ctrl.Idx)
            self:Notify("Copied flag to clipboard!")
            menu:Hide()
        end)
    end

    self:AddToRegistry(frame, { BackgroundColor3 = "MainColor", BorderColor3 = "OutlineColor" })
    self:AddToRegistry(fill, { BackgroundColor3 = "AccentColor" })

    ctrl:UpdateDisplay()
    return ctrl
end

-- Dropdown (full implementation)
function Lib:CreateDropdown(parent, config)
    config = config or {}
    local ctrl = {
        Type = "Dropdown",

        Multi = config.Multi or false,
        Value = config.Default or (config.Multi and {} or nil),

        AllowNull = config.AllowNull or false,
        Illegal = config.Illegal or false,
    }
    self.Controls[ctrl.Idx] = ctrl

    local frame = Instance.new("Frame")
    frame.Parent = parent
    frame.BackgroundColor3 = self.MainColor
    frame.BorderColor3 = self.OutlineColor
    frame.BorderMode = Enum.BorderMode.Middle
    frame.Size = UDim2.new(1, -2, 0, 28)
    frame.ZIndex = 1
    frame.Active = true

    label.Parent = frame
    label.BackgroundTransparency = 1
    label.Text = config.Text or "Dropdown"
    label.Font = self.Font
    label.TextColor3 = self.FontColor

    label.Position = UDim2.new(0, 5, 0, 0)
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.ZIndex = 2

    arrow.TextColor3 = self.FontColor
    arrow.TextSize = 14

    arrow.Position = UDim2.new(1, -24, 0, 0)
    arrow.TextXAlignment = Enum.TextXAlignment.Center

    dropFrame.Parent = parent

    dropFrame.BorderColor3 = self.OutlineColor
    dropFrame.BorderMode = Enum.BorderMode.Inset
    dropFrame.Size = UDim2.new(0, 0, 0, 0)
    dropFrame.ZIndex = 20
    dropFrame.Visible = false
    dropFrame.ClipsDescendants = true

    local scroll = Instance.new("ScrollingFrame")
    scroll.Parent = dropFrame
    scroll.BackgroundTransparency = 1
    scroll.BorderSizePixel = 0
    scroll.Size = UDim2.new(1, 0, 1, 0)
    scroll.ZIndex = 21
    scroll.ScrollingDirection = Enum.ScrollingDirection.Y
    scroll.ScrollBarThickness = 3
    scroll.CanvasSize = UDim2.new(0, 0, 0, 0)

    local layout = Instance.new("UIListLayout")
    layout.Parent = scroll
    layout.FillDirection = Enum.FillDirection.Vertical
    layout.SortOrder = Enum.SortOrder.LayoutOrder
    layout.Padding = UDim.new(0, 2)

    local optionButtons = {}

    function ctrl:BuildDropdownList()
        for _, child in ipairs(scroll:GetChildren()) do
            if child:IsA("Frame") then child:Destroy() end
        end
        optionButtons = {}
        for _, val in ipairs(self.Values) do
            local btn = Instance.new("Frame")
            btn.Parent = scroll
            btn.BackgroundColor3 = self.MainColor
            btn.BorderColor3 = self.OutlineColor
            btn.BorderMode = Enum.BorderMode.Middle
            btn.Size = UDim2.new(1, -2, 0, 28)
            btn.ZIndex = 22
            btn.Active = true

            local lbl = Instance.new("TextLabel")
            lbl.Parent = btn
            lbl.BackgroundTransparency = 1
            lbl.Text = val
            lbl.Font = self.Font
            lbl.TextColor3 = self.FontColor
            lbl.TextSize = 14
            lbl.Size = UDim2.new(1, -6, 1, 0)
            lbl.Position = UDim2.new(0, 6, 0, 0)
            lbl.TextXAlignment = Enum.TextXAlignment.Left
            lbl.ZIndex = 23

            local isSelected = false
            if self.Multi then
                isSelected = self.Value[val] or false
            else
                isSelected = self.Value == val
            end

            function btn:Update()
                local sel = self.Multi and self.Value[val] or self.Value == val
                lbl.TextColor3 = sel and self.AccentColor or self.FontColor
            end

            btn.MouseButton1Click:Connect(function()
                if self.Multi then
                    local new = not self.Value[val]
                    if new then
                        self.Value[val] = true
                    else
                        self.Value[val] = nil
                    end
                else
                    if not self.AllowNull and self.Value == val then
                        return
                    end
                    self.Value = (self.Value == val and self.AllowNull and nil) or val
                end
                self:BuildDropdownList()
                self:UpdateDisplay()
                for _, func in ipairs(self.ChangedFuncs) do
                    func(self.Value)
                end
                self.Callback(self.Value)
                Lib.SaveManager:Save(nil, true)
            end)

            optionButtons[val] = btn
        end
        scroll.CanvasSize = UDim2.new(0, 0, 0, #self.Values * 30 + 4)
        local maxHeight = 200
        local height = math.min(#self.Values * 30 + 4, maxHeight)
        dropFrame.Size = UDim2.new(0, frame.AbsoluteSize.X + 2, 0, height)
        dropFrame.Position = UDim2.new(0, frame.AbsolutePosition.X - parent.AbsolutePosition.X - 1,
                                       0, frame.AbsolutePosition.Y - parent.AbsolutePosition.Y + frame.Size.Y.Offset + 1)
    end

    function ctrl:UpdateDisplay()
        local text = ""
        if self.Multi then
            local parts = {}
            for k, v in pairs(self.Value) do
                if v then table.insert(parts, k) end
            end
            text = table.concat(parts, ", ")
        else
            text = self.Value or "..."
        end
        label.Text = text == "" and "..." or text
    end

    function ctrl:SetValue(value)
        if self.Multi then
            local new = {}
            for k, v in pairs(value) do
                if self.Illegal or table.find(self.Values, k) then
                    new[k] = true
                end
            end
            self.Value = new
        else
            if value and (self.Illegal or table.find(self.Values, value)) then
                self.Value = value
            else
                self.Value = nil
            end
        end
        self:BuildDropdownList()
        self:UpdateDisplay()
        for _, func in ipairs(self.ChangedFuncs) do
            func(self.Value)
        end
        self.Callback(self.Value)
        Lib.SaveManager:Save(nil, true)
    end

    function ctrl:OnChanged(func)
        table.insert(self.ChangedFuncs, func)
        func(self.Value)
    end

    function ctrl:Remove()
        frame:Destroy()
        dropFrame:Destroy()
        self.ChangedFuncs = nil
        Lib.Controls[ctrl.Idx] = nil
    end

    frame.MouseButton1Click:Connect(function()
        if dropFrame.Visible then
            dropFrame.Visible = false
            self.OpenedFrames[dropFrame] = nil
        else
            self:BuildDropdownList()
            dropFrame.Visible = true
            self.OpenedFrames[dropFrame] = true
        end
    end)

    UserInputService.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            local mouse = UserInputService:GetMouseLocation()
            local pos = dropFrame.AbsolutePosition
            local size = dropFrame.AbsoluteSize
            if not (mouse.X >= pos.X and mouse.X <= pos.X + size.X and
                    mouse.Y >= pos.Y and mouse.Y <= pos.Y + size.Y) then
                dropFrame.Visible = false
                self.OpenedFrames[dropFrame] = nil
            end
        end
    end)

    self:AddToRegistry(frame, { BackgroundColor3 = "MainColor", BorderColor3 = "OutlineColor" })
    ctrl:BuildDropdownList()
    ctrl:UpdateDisplay()
    return ctrl
end

function Lib:AddContextMenu(trigger, parentFrame)
    local menu = {}
    menu.Options = {}
    menu.Visible = false

    local container = Instance.new("Frame")
    container.Parent = self.ScreenGui
    container.BackgroundTransparency = 1
    container.Size = UDim2.new(0, 0, 0, 0)
    container.ZIndex = 14
    container.Visible = false

    local inner = Instance.new("Frame")
    inner.Parent = container
    inner.BackgroundColor3 = self.BackgroundColor
    inner.BorderColor3 = self.OutlineColor
    inner.BorderMode = Enum.BorderMode.Inset
    inner.Size = UDim2.new(1, 0, 1, 0)
    inner.ZIndex = 15

    local layout = Instance.new("UIListLayout")
    layout.Parent = inner
    layout.FillDirection = Enum.FillDirection.Vertical
    layout.SortOrder = Enum.SortOrder.LayoutOrder
    layout.Padding = UDim.new(0, 2)

    local function reposition()
        local pos = trigger.AbsolutePosition
        container.Position = UDim2.new(0, pos.X + trigger.AbsoluteSize.X + 4, 0, pos.Y + 1)
        local maxWidth = 0
        local totalHeight = 0
        for _, child in ipairs(inner:GetChildren()) do
            if child:IsA("TextLabel") then
                maxWidth = math.max(maxWidth, child.TextBounds.X)
                totalHeight = totalHeight + 22
            end
        end
        container.Size = UDim2.new(0, maxWidth + 12, 0, totalHeight + 6)
    end

    function menu:Show()
        reposition()
        container.Visible = true
        self.Visible = true
        Lib.OpenedFrames[container] = true
    end

    function menu:Hide()
        container.Visible = false
        self.Visible = false
        Lib.OpenedFrames[container] = nil
    end

    function menu:AddOption(text, callback)
        local lbl = Instance.new("TextLabel")
        lbl.Parent = inner
        lbl.BackgroundTransparency = 1
        lbl.Text = text
        lbl.Font = Lib.Font
        lbl.TextColor3 = Lib.FontColor
        lbl.TextSize = 13
        lbl.Size = UDim2.new(1, 0, 0, 22)
        lbl.ZIndex = 16
        lbl.Active = true
        lbl.TextXAlignment = Enum.TextXAlignment.Center

        lbl.MouseButton1Click:Connect(function()
            if callback then callback() end
            menu:Hide()
        end)

        lbl.MouseEnter:Connect(function()
            lbl.TextColor3 = Lib.AccentColor
        end)
        lbl.MouseLeave:Connect(function()
            lbl.TextColor3 = Lib.FontColor
        end)

        table.insert(menu.Options, lbl)
        return lbl
    end

    local closeConn
    closeConn = UserInputService.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            local mouse = UserInputService:GetMouseLocation()
            local pos = container.AbsolutePosition
            local size = container.AbsoluteSize
            if not (mouse.X >= pos.X and mouse.X <= pos.X + size.X and
                    mouse.Y >= pos.Y and mouse.Y <= pos.Y + size.Y) then
                menu:Hide()
            end
        end
    end)
    table.insert(Lib.Signals, closeConn)

    trigger.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton2 then
            menu:Show()
        end
    end)

    return menu
end

function Lib:AddDependencyBox(container, dependencies)
    local box = {}
    box.Container = container
    box.Dependencies = dependencies
    box.Visible = true

    function box:Update()
        local visible = true
        for _, dep in ipairs(self.Dependencies) do
            if dep[1].Value ~= dep[2] then
                visible = false
                break
            end
        end
        self.Container.Visible = visible
        if self.Container.Resize then self.Container:Resize() end
    end

    box:Update()
    table.insert(self.DependencyBoxes, box)
    return box
end

-- ============================================================
-- Registry for dynamic theme updates
-- ============================================================
function Lib:AddToRegistry(instance, properties, isHud)
    local entry = {
        Instance = instance,
        Properties = properties,
    }
    table.insert(self.Registry, entry)
    self.RegistryMap[instance] = entry
    if isHud then table.insert(self.HudRegistry, entry) end
end

function Lib:RemoveFromRegistry(instance)
    local entry = self.RegistryMap[instance]
    if entry then
        for i, e in ipairs(self.Registry) do
            if e == entry then table.remove(self.Registry, i) break end
        end
        for i, e in ipairs(self.HudRegistry) do
            if e == entry then table.remove(self.HudRegistry, i) break end
        end
        self.RegistryMap[instance] = nil
    end
end

function Lib:UpdateColorsUsingRegistry()
    for _, entry in ipairs(self.Registry) do
        local inst = entry.Instance
        for prop, value in pairs(entry.Properties) do
            if type(value) == "string" then
                inst[prop] = self[value] or value
            elseif type(value) == "function" then
                inst[prop] = value()
            end
        end
    end
end

-- MakeDraggable (simplified)
-- ============================================================
function Lib:MakeDraggable(frame, topBarHeight, condition)
    frame.Active = true
    local dragging = false
    local dragInput
    local dragStart
    local startPos

    frame.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            if condition and self.CantDragForced then return end
            local localY = input.Position.Y - frame.AbsolutePosition.Y
            if localY > (topBarHeight or 40) then return end
            dragging = true
            dragInput = input
            dragStart = input.Position
            startPos = frame.Position
        end
    end)

    frame.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if not dragging or input ~= dragInput then return end
        local delta = input.Position - dragStart
        frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input == dragInput or input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
            dragInput = nil
        end
    end)
end

        mainWindow:Resize()

-- Aimbot settings
local aimDeadCheck = true
local aimWallCheck = true
local aimPriority = "Distance"
local aimFOV = 500
local aimMaxDistance = 0

local function isValidTarget(player)
	if player == Player or not player.Character or not player.Character:FindFirstChild("Humanoid") or not player.Character:FindFirstChild("Head") then 
		return false 
	end
	
	if aimDeadCheck and player.Character.Humanoid.Health <= 0 then 
		return false 
	end
	
	if aimMaxDistance > 0 then
		local myRoot = getRoot()
		if myRoot then
			local distance = (myRoot.Position - player.Character.Head.Position).Magnitude
			if distance > aimMaxDistance then
				return false
			end
		end
	end
	
	if aimWallCheck then
		local ray = Ray.new(Camera.CFrame.Position, (player.Character.Head.Position - Camera.CFrame.Position).Unit * 500)
		local part = Workspace:FindPartOnRayWithIgnoreList(ray, {getCharacter()})
		if part and part:IsDescendantOf(player.Character) then 
			return true 
		else 
			return false 
		end
	end
	return true
end

local function getClosestPlayer(fov)
	local closest, closestValue = nil, math.huge
	local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
	
	for _, player in ipairs(Players:GetPlayers()) do
		if not isValidTarget(player) then continue end
		local head = player.Character.Head
		local pos, onScreen = Camera:WorldToViewportPoint(head.Position)
		if not onScreen then continue end
		local dist = (center - Vector2.new(pos.X, pos.Y)).Magnitude
		if dist > (fov or math.huge) then continue end
		local value = aimPriority == "Distance" and dist or player.Character.Humanoid.Health
		if value < closestValue then
			closestValue = value
			closest = player
		end
	end
	return closest
end

-- Aimbot loop
connections["Aimbot"] = RunService:BindToRenderStep("Aimbot", Enum.RenderPriority.Camera.Value, function(dt)
	if not aimbotEnabled then return end
	local closest = getClosestPlayer(aimFOV)
	if closest and closest.Character then
		local targetPart = closest.Character:FindFirstChild(aimbotLock == "Head" and "Head" or "UpperTorso")
		if not targetPart then return end
		
		local prediction = aimbotPrediction and targetPart.AssemblyLinearVelocity * 0.1 or Vector3.zero
		local targetPos = targetPart.Position + prediction
		
		if aimbotMode == "Rage" then
			Camera.CFrame = CFrame.lookAt(Camera.CFrame.Position, targetPos)
		elseif aimbotMode == "Legit" then
			local currentLook = Camera.CFrame.LookVector
			local targetLook = (targetPos - Camera.CFrame.Position).Unit
			local smoothFactor = math.clamp(aimbotSensitivity * dt * 10, 0.01, 0.3)
			local newLook = currentLook:Lerp(targetLook, smoothFactor)
			Camera.CFrame = CFrame.lookAt(Camera.CFrame.Position, Camera.CFrame.Position + newLook)
		end
		
		if aimbotAutoFire then
			local tool = Player.Character and Player.Character:FindFirstChildOfClass("Tool")
			if tool then
				tool:Activate()
			end
		end
	end
end)

		local args = {...}
		if silentAimEnabled and #args >= 3 and math.random(100) <= silentAimHitchance then
			local closest = getClosestPlayer(aimFOV)
			if closest and closest.Character and closest.Character:FindFirstChild("Head") then
				args[3] = closest.Character.Head.Position
			end
		end
		return oldRaycast(table.unpack(args))
	end
end)

-- ESP storage and functions
local espBoxes = {}
local espChams = {}
local espTexts = {}

local function cleanupPlayerESP(player)
	if espBoxes[player] then 
		pcall(function() espBoxes[player]:Remove() end) 
		espBoxes[player] = nil 
	end
	if espChams[player] then 
		pcall(function() espChams[player]:Destroy() end) 
		espChams[player] = nil 
	end
	if espTexts[player] then 
		pcall(function() espTexts[player]:Remove() end) 
		espTexts[player] = nil 
	end
end

local function createPlayerESP(player)
	if player == Player then return end
	if not player.Character then return end
	
	cleanupPlayerESP(player)
	
	local box = Drawing.new("Square")
	box.Visible = false
	box.Color = Color3.new(1, 1, 1)
	box.Thickness = 2
	box.Filled = false
	box.Transparency = 1
	espBoxes[player] = box
	
	local cham = Instance.new("Highlight")
	cham.Name = "ESP_Cham_" .. player.Name
	cham.Parent = game.CoreGui
	cham.Adornee = player.Character
	cham.FillTransparency = 0.5
	cham.OutlineTransparency = 0
	cham.FillColor = Color3.new(1, 1, 1)
	cham.OutlineColor = Color3.new(1, 1, 1)
	cham.Enabled = false
	espChams[player] = cham
	
	local text = Drawing.new("Text")
	text.Visible = false
	text.Color = Color3.new(1, 1, 1)
	text.Size = 16
	text.Center = true
	text.Outline = true
	text.Transparency = 1
	text.Font = 2
	espTexts[player] = text
end

-- Player management
local function setupPlayerESP(player)
	if player == Player then return end
	
	player.CharacterAdded:Connect(function(char)
		task.wait(0.1)
		if playerESPEnabled then 
			createPlayerESP(player)
		end
	end)
	
	player.CharacterRemoving:Connect(function()
		cleanupPlayerESP(player)
	end)
	
	if player.Character then
		task.wait(0.1)
		createPlayerESP(player)
	end
end

for _, player in ipairs(Players:GetPlayers()) do
	setupPlayerESP(player)
end

Players.PlayerAdded:Connect(setupPlayerESP)
Players.PlayerRemoving:Connect(cleanupPlayerESP)

-- ESP update loop
connections["PlayerESP"] = RunService.Heartbeat:Connect(function()
	if not playerESPEnabled then return end
	
	for _, player in ipairs(Players:GetPlayers()) do
		if player == Player then continue end
		
		if not player.Character or not player.Character.Parent then
			cleanupPlayerESP(player)
			continue
		end
		
		local humanoid = player.Character:FindFirstChild("Humanoid")
		local root = player.Character:FindFirstChild("HumanoidRootPart")
		
		if not humanoid or not root then 
			cleanupPlayerESP(player)
			continue 
		end
		
		if not espBoxes[player] then
			createPlayerESP(player)
		end
		
		local rootPos, onScreen = Camera:WorldToViewportPoint(root.Position)
		if onScreen then
			if espTexts[player] then
				local textStr = ""
				if playerESPName then textStr = textStr .. player.Name .. "\n" end
				if playerESPDistance then 
					local myRoot = getRoot()
					if myRoot then
						textStr = textStr .. math.floor((myRoot.Position - root.Position).Magnitude) .. " studs\n" 
					end
				end
				if playerESPHealth then textStr = textStr .. math.floor(humanoid.Health) .. "/" .. humanoid.MaxHealth .. "\n" end
				if playerESPWeapon then
					local tool = player.Character:FindFirstChildOfClass("Tool") or player.Backpack:FindFirstChildOfClass("Tool")
					if tool then textStr = textStr .. tool.Name end
				end
				
				espTexts[player].Text = textStr
				espTexts[player].Position = Vector2.new(rootPos.X, rootPos.Y - 50)
				espTexts[player].Visible = true
			end
			
			if boxESPEnabled and espBoxes[player] then
				local minX, minY = math.huge, math.huge
				local maxX, maxY = -math.huge, -math.huge
				
				for _, part in ipairs(player.Character:GetDescendants()) do
					if part:IsA("BasePart") then
						local corners = {
							part.CFrame * CFrame.new(part.Size.X/2, part.Size.Y/2, part.Size.Z/2).Position,
							part.CFrame * CFrame.new(part.Size.X/2, part.Size.Y/2, -part.Size.Z/2).Position,
							part.CFrame * CFrame.new(part.Size.X/2, -part.Size.Y/2, part.Size.Z/2).Position,
							part.CFrame * CFrame.new(part.Size.X/2, -part.Size.Y/2, -part.Size.Z/2).Position,
							part.CFrame * CFrame.new(-part.Size.X/2, part.Size.Y/2, part.Size.Z/2).Position,
							part.CFrame * CFrame.new(-part.Size.X/2, part.Size.Y/2, -part.Size.Z/2).Position,
							part.CFrame * CFrame.new(-part.Size.X/2, -part.Size.Y/2, part.Size.Z/2).Position,
							part.CFrame * CFrame.new(-part.Size.X/2, -part.Size.Y/2, -part.Size.Z/2).Position
						}
						for _, corner in ipairs(corners) do
							local pos = Camera:WorldToViewportPoint(corner)
							minX = math.min(minX, pos.X)
							maxX = math.max(maxX, pos.X)
							minY = math.min(minY, pos.Y)
							maxY = math.max(maxY, pos.Y)
						end
					end
				end
				
				espBoxes[player].Size = Vector2.new(maxX - minX, maxY - minY)
				espBoxes[player].Position = Vector2.new(minX, minY)
				espBoxes[player].Visible = true
			end
		else
			if espTexts[player] then espTexts[player].Visible = false end
			if espBoxes[player] then espBoxes[player].Visible = false end
		end
		
		if chamsESPEnabled and espChams[player] then
			if espChams[player].Adornee ~= player.Character then
				espChams[player].Adornee = player.Character
			end
			espChams[player].Enabled = true
		elseif espChams[player] then
			espChams[player].Enabled = false
		end
	end
end)

local function setSpeed(val)
    speedValue = val

    if speedConn then
        speedConn:Disconnect()
        speedConn = nil
    end

    local humanoid = getHumanoid()
    if humanoid then
        humanoid.WalkSpeed = val
        speedConn = humanoid:GetPropertyChangedSignal("WalkSpeed"):Connect(function()
            if speedEnabled and humanoid.WalkSpeed ~= speedValue then
                humanoid.WalkSpeed = speedValue
            end
        end)
    end
end

    Name = "Enable Speed",
    Default = false,
    Callback = function(Value)
        speedEnabled = Value
        if Value then
            setSpeed(speedValue)
        else
            if speedConn then
                speedConn:Disconnect()
                speedConn = nil
            end
            local humanoid = getHumanoid()
            if humanoid then
                humanoid.WalkSpeed = 16
            end
        end
    end
})

    Max = 500,
    Default = 16,

    Callback = function(Value)

function unloadScript()
	print("Unloading arch script free...")
	Library.Unloaded = true
	
	-- Disconnect all connections
	for name, conn in pairs(connections) do
		pcall(function()
			if name == "Aimbot" then
				RunService:UnbindFromRenderStep("Aimbot")
			elseif type(conn) == "table" and conn.Disconnect then
				conn:Disconnect()
			end
		end)
	end
	connections = {}
	
	-- Clean ESP
	for player in pairs(espBoxes) do
		cleanupPlayerESP(player)
	end
	
	-- Remove FOV circle
	pcall(function() fovCircle:Remove() end)
	
	-- Reset physics
	if flyBodyVelocity then flyBodyVelocity:Destroy() end
	Workspace.Gravity = 196.2
	
	-- Reset camera
	if originalCameraType then
		Player.CameraMode = originalCameraType
	else
		Player.CameraMode = Enum.CameraMode.Classic
	end
	if originalMaxZoom then
		Player.CameraMaxZoomDistance = originalMaxZoom
	else
		Player.CameraMaxZoomDistance = 128
	end
	if originalMinZoom then
		Player.CameraMinZoomDistance = originalMinZoom
	else
		Player.CameraMinZoomDistance = 0.5
	end
	
	-- Reset humanoid
	pcall(function()
		local humanoid = getHumanoid()
		if humanoid then
			humanoid.WalkSpeed = 16
			humanoid.JumpPower = 50
		end
	end)
	
	-- 清理 Speed 監聽
	if speedConn then
		speedConn:Disconnect()
		speedConn = nil
	end
	speedEnabled = false
	
	-- Disconnect infinite jump
	if infJumpConnection then infJumpConnection:Disconnect() end
	
	-- Reset lighting
	pcall(function()
		local lighting = game:GetService("Lighting")
		lighting.GlobalShadows = true
		lighting.Brightness = 1
		settings().Rendering.QualityLevel = Enum.QualityLevel.Automatic
	end)
	
	-- Destroy UI
	ScreenGui:Destroy()
	
	Library:Notify("Script unloaded successfully", "success")
end
