genv.AXIS_TELEPORT_RESUME = nil
do
    local HttpService = game:GetService("HttpService")
    local function canFile()
        return type(writefile) == "function" and type(readfile) == "function" and type(isfile) == "function"
    end
    local function enc(v)
        if typeof(v) == "Color3" then return { __c3 = { v.R, v.G, v.B } } end
        return v
    end
    local function dec(v)
        if type(v) == "table" and type(v.__c3) == "table" then return Color3.new(v.__c3[1], v.__c3[2], v.__c3[3]) end
        return v
    end

    function Persist.save(markTeleport)
        if not canFile() then return false end
        local data = {}
        for k, v in pairs(AXIS) do
            local t = typeof(v)
            if t == "boolean" or t == "number" or t == "string" or t == "Color3" or t == "table" then data[k] = enc(v) end
        end
        local payload = { settings = data, teleportAt = markTeleport and os.time() or nil }
        return pcall(function()
            if type(isfolder) == "function" and type(makefolder) == "function" and not isfolder("AXIS") then makefolder("AXIS") end
            writefile(Persist.file, HttpService:JSONEncode(payload))
        end)
    end

    local function readState()
        if not canFile() then return nil end
        local ok, d = pcall(function()
            if not isfile(Persist.file) then return nil end
            return HttpService:JSONDecode(readfile(Persist.file))
        end)
        return ok and type(d) == "table" and d or nil
    end

    function Persist.apply(d)
        local settings = type(d.settings) == "table" and d.settings or d
        Persist.pending = {}
        for k, v in pairs(settings) do
            v = dec(v)
            if type(v) == "boolean" then
                AXIS[k] = false
                if v then Persist.pending[k] = true end
            elseif k ~= "teleportAt" then
                AXIS[k] = v
            end
        end
    end

    local function queueFn()
        if type(queue_on_teleport) == "function" then return queue_on_teleport end
        if type(queueonteleport) == "function" then return queueonteleport end
        if type(queueteleport) == "function" then return queueteleport end
        if type(syn) == "table" and type(syn.queue_on_teleport) == "function" then return syn.queue_on_teleport end
        if type(fluxus) == "table" and type(fluxus.queue_on_teleport) == "function" then return fluxus.queue_on_teleport end
        return nil
    end

    function Persist.loaderCode()
        local url = genv.AXIS_SCRIPT_URL
        if type(url) == "string" and url ~= "" then
            return string.format("getgenv().AXIS_TELEPORT_RESUME = true getgenv().AXIS_SCRIPT_URL = %q loadstring(game:HttpGet(%q))()", url, url)
        end
        local p1 = genv.AXIS_SCRIPT_PATH or "rivals_auto_kill_v26.lua"
        local p2 = "AXIS/rivals_auto_kill_v26.lua"
        return string.format([[
getgenv().AXIS_TELEPORT_RESUME = true
getgenv().AXIS_SCRIPT_PATH = %q
local paths = { %q, %q }
for _, p in ipairs(paths) do
    if isfile and isfile(p) then
        local f, e = loadstring(readfile(p))
        if f then return f() end
        warn("[axis] yukleme hatasi: " .. tostring(e)) return
    end
end
warn("[axis] Auto Execute: script dosyasi bulunamadi -> executor workspace klasorune " .. %q .. " olarak kaydedin")
]], p1, p1, p2, p1)
    end

    function Persist.queue()
        if Persist.queued then return true end
        local q = queueFn()
        if not q then warn("[axis] Auto Execute: executor queue_on_teleport desteklemiyor") return false end
        local ok = pcall(q, Persist.loaderCode())
        Persist.queued = ok
        return ok
    end

        end

        Persist.resuming = false
    end

    if GameState == newState then return end
    local old = GameState
    GameState, StateStart = newState, tick()
    if VOID_STATES[newState] then voidEntryTime = tick() end
    teleportDone, teleportAt = false, 0

        local hrp = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
        waitAnchor = hrp and hrp.Position or nil
    end
    log("STATE: " .. old .. " -> " .. newState)
end

local RealHealth = { hp = nil, max = nil }
local function getRealHealth(hum)
    if AXIS.GodModeEnabled and RealHealth.max then return RealHealth.hp or RealHealth.max, RealHealth.max end
    return hum.Health, hum.MaxHealth
end

local PhysOriginals = setmetatable({}, { __mode = "k" })
local function applyPhysicalImmunity(part)
    if AXIS.GodModeEnabled and AXIS.PhysicalImmunity and part:IsA("BasePart") and not PhysOriginals[part] then
        PhysOriginals[part] = { CanCollide = part.CanCollide, Props = part.CustomPhysicalProperties }
        part.CanCollide = false
        part.CustomPhysicalProperties = PhysicalProperties.new(0.01, 0, 0, 0, 0)
    end
end
local function revertPhysicalImmunity()
    for part, o in pairs(PhysOriginals) do
        if typeof(part) == "Instance" then
            pcall(function() part.CanCollide = o.CanCollide part.CustomPhysicalProperties = o.Props end)
        end
        PhysOriginals[part] = nil
    end
end

local godApplied = false
local function applyGodMode(char, hum)
    if hum.MaxHealth < 1e8 then RealHealth.max = hum.MaxHealth RealHealth.hp = hum.Health end
    pcall(function() hum.MaxHealth = 1e9 hum.Health = hum.MaxHealth end)
    if AXIS.ForceFieldEnabled and not char:FindFirstChild("AXIS_FF") then
        local ff = Instance.new("ForceField")
        ff.Name, ff.Visible, ff.Parent = "AXIS_FF", false, char
    elseif not AXIS.ForceFieldEnabled and char:FindFirstChild("AXIS_FF") then
        char.AXIS_FF:Destroy()
    end
    if AXIS.PhysicalImmunity then
        -- tam tarama sadece karakter basina bir kez; yeni parcalari DescendantAdded yakalar
        if PhysOriginals.__char ~= char then
            rawset(PhysOriginals, "__char", char)
            for _, d in ipairs(char:GetDescendants()) do applyPhysicalImmunity(d) end
        end
    else
        revertPhysicalImmunity()
    end
    if not godApplied then godApplied = true log("God Mode aktif: " .. char.Name) end
end
local function removeGodMode(char, hum)
    if not godApplied then return end
    godApplied = false
    if RealHealth.max then
        pcall(function()
            hum.MaxHealth = RealHealth.max
            hum.Health = math.min(RealHealth.hp or RealHealth.max, RealHealth.max)
        end)
    end
    local ff = char:FindFirstChild("AXIS_FF")
    if ff then ff:Destroy() end
    revertPhysicalImmunity()
    log("God Mode kapandi")
end

local function setupGodMode(char)
    if not char then return end
    task.spawn(function()
        local hum = char:WaitForChild("Humanoid", 10)
        if not hum then return end
        godApplied = false
        if hum.MaxHealth < 1e8 then RealHealth.max = hum.MaxHealth RealHealth.hp = hum.Health end

        track(hum.HealthChanged:Connect(function(health)
            if health < 1e8 then RealHealth.hp = health end
            if AXIS.GodModeEnabled and health < hum.MaxHealth and health > 0 then
                hum.Health = hum.MaxHealth
            end
        end))

        pcall(function()
            local oldTakeDamage = hum.TakeDamage
            hum.TakeDamage = function(self, amount)
                if AXIS.GodModeEnabled then return end
                return oldTakeDamage(self, amount)
            end
        end)

        track(char.DescendantAdded:Connect(applyPhysicalImmunity))
    end)
end

track(lp.CharacterAdded:Connect(function(char)
    setGameState("INIT_VOID")
    setupGodMode(char)
end))
if lp.Character then setupGodMode(lp.Character) end

local SPAWN_IMMUNE_TAGS = { "SpawnProtection", "SpawnShield", "Immune", "Invincible",
    "Protected", "SpawnImmune", "GodMode", "SpawnImmunity" }
local CHAR_ATTRS = { "SpawnProtection", "Protected", "Invincible", "IsInvincible",
    "SpawnShield", "Immune", "SpawnImmune", "GodMode", "ImmuneUntil" }
local HUM_ATTRS  = { "SpawnProtection", "Protected", "Invincible", "IsInvincible", "Immune", "Spawning" }
local PLR_ATTRS  = { "SpawnProtection", "Protected", "Invincible", "IsInvincible", "Immune", "SpawnShield" }

local function computeProtection(char)
    if char:FindFirstChildWhichIsA("ForceField", true) then return true end

    for _, a in ipairs(CHAR_ATTRS) do
        local v = char:GetAttribute(a)
        if v == true or (typeof(v) == "number" and v > tick()) then return true end
    end

    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum then
        for _, a in ipairs(HUM_ATTRS) do
            if hum:GetAttribute(a) == true then return true end
        end
        local st = hum:GetState()
        if st == Enum.HumanoidStateType.GettingUp or st == Enum.HumanoidStateType.Physics then return true end
    end

    local plr = Players:GetPlayerFromCharacter(char)
    if plr then
        for _, a in ipairs(PLR_ATTRS) do
            if plr:GetAttribute(a) == true then return true end
        end
    end

    for _, tag in ipairs(SPAWN_IMMUNE_TAGS) do
        if CollectionService:HasTag(char, tag) or (hum and CollectionService:HasTag(hum, tag)) then return true end
    end

    for _, c in ipairs(char:GetChildren()) do
        local nm = c.Name:lower()
        if nm:find("immune") or nm:find("invincible")
           or (nm:find("spawn") and (nm:find("shield") or nm:find("protect"))) then
            return true
        end
    end

    if plr then
        local ok, fighter = pcall(function() return FC:GetFighter(plr) end)
        if ok and fighter then
            if fighter.Entity then
                local okE, inv = pcall(function() return fighter.Entity:Get("IsInvincible") end)
                if okE and inv == true then return true end
            end
            local d = fighter.Data
            if d and (d.IsInvincible == true or d.SpawnProtection == true) then return true end
        end
    end
    return false
end

local protCache = setmetatable({}, { __mode = "k" })   -- karakter -> {t, v}
local function hasProtection(char)
    if not char then return true end
    local now, e = tick(), protCache[char]
    if e and now - e.t < (AXIS.ProtectionCacheTime or 0.25) then return e.v end
    local ok, v = pcall(computeProtection, char)
    v = (not ok) or v
    protCache[char] = { t = now, v = v }
    return v
end

local function matchesWord(name, kw)
    -- " name " icinde " kw " arar; bosluk/alt cizgi/tire ayirici sayilir
    local padded = " " .. name:lower():gsub("[%p_]", " ") .. " "
    return padded:find(" " .. kw .. " ", 1, true) ~= nil
end
local function matchesAny(name, list)
    if not name or name == "" then return false end
    for _, kw in ipairs(list) do if matchesWord(name, kw) then return true end end
    return false
end

local function getEnemyEquippedItemName(plr)

        return tostring(fighter.EquippedItem.Name or "")
    end
    local tool = plr.Character and plr.Character:FindFirstChildOfClass("Tool")
    return tool and tool.Name or ""
end

local function aliveHumanoid(char)
    local hum = char and char:FindFirstChildOfClass("Humanoid")

local function getClosestEnemy()
    local myHrp = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
    if not myHrp then return nil end
    local best, bestDist = nil, math.huge
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= lp and not isTeammate(plr) then
            local char = plr.Character
            if char and aliveHumanoid(char) and not hasProtection(char) then
                local ref = char:FindFirstChild("HitboxHead") or char:FindFirstChild("Head")
                    or char:FindFirstChild("HumanoidRootPart")
                if ref then
                    local d = (myHrp.Position - ref.Position).Magnitude
                    if d < bestDist then best, bestDist = plr, d end
                end
            end
        end
    end
    return best, bestDist
end

local function hasAnyLiveEnemy()
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= lp and not isTeammate(plr) and aliveHumanoid(plr.Character) then return true end
    end
    return false
end

local MUZZLE_CF = CFrame.new(0.43, 0.25, 0.42)

local function fireAt(targetPart)

    local item = equippedItem()
    local oid = itemGet(item, "ObjectID")
    if not oid then return end

    setSilentTarget(targetPos)

    local aimCF = encodeCF(CFrame.new(shootPos, targetPos))
    local muzzle = encodeCF(MUZZLE_CF)
    local data = {
        [utf8.char(1)] = {
            [utf8.char(0)] = aimCF,
            [utf8.char(1)] = aimCF,
            [utf8.char(2)] = targetPart,
            [utf8.char(3)] = muzzle,
        },
    }
    if AXIS.ModNoSpread then data[utf8.char(2)] = true end   -- No Spread (Kicia: args["\2"] = true)
    for _ = 1, math.clamp(AXIS.BurstCount or 4, 1, 10) do
        safeCall("FireServer", function() UseItemRemote:FireServer(oid, START_SHOOTING, data, nil) end)
    end
    stats.Shots += 1
end

local function validPos(pos)
    return pos and pos.X == pos.X and pos.Y == pos.Y and pos.Z == pos.Z and pos.Y >= CFG.SAFE_MIN_Y
end

local function randomVoidPos()
    local M = AXIS.VoidMaxValue or 1000000
    local minY = math.max(AXIS.VoidMinY or 5000, CFG.SAFE_MIN_Y + 1)
    -- Eski surumde Y negatif cikinca safeCFrame nil donuyor, karelerin ~yarisi bosa gidiyordu
    return Vector3.new(math.random(-M, M), math.random(minY, math.max(minY, M)), math.random(-M, M))
end

local function goVoid(hrp)
    hrp.CFrame = CFrame.new(randomVoidPos())
    hrp.AssemblyLinearVelocity  = Vector3.zero
    hrp.AssemblyAngularVelocity = Vector3.zero
end

-- Heartbeat icinde yield yok (eski Heartbeat:Wait() kareleri ust uste bindiriyordu)
local function serverSafeTeleport(hrp, pos, lookAt)
    if not hrp or not validPos(pos) then return false end
    hrp.AssemblyLinearVelocity  = Vector3.zero
    hrp.AssemblyAngularVelocity = Vector3.zero
    hrp.CFrame = lookAt and CFrame.new(pos, lookAt) or CFrame.new(pos)
    local hum = hrp.Parent and hrp.Parent:FindFirstChildOfClass("Humanoid")
    if hum then pcall(function() hum:ChangeState(Enum.HumanoidStateType.Physics) end) end
    return true
end

-- ============================================================
-- HVH ANTI-AIM (mutlak acilar: kareler arasi birikme yok)
-- Yaw: None / Spin / Jitter / Random / Backwards / Sideways / Spin+Jitter
-- Pitch: None / Up / Down / Random / Custom   Roll: None / Upside Down / Random
-- + ek yaw jitter + karakter pozisyon jitter
-- ============================================================
local TAU = math.pi * 2

local function antiAimTick(dt)
    antiAimAngle = (antiAimAngle + math.rad(clampNum(AXIS.SpinBypassSpeed, 0, 7200, 720)) * dt) % TAU
    jitterTimer += dt
    if jitterTimer >= clampNum(AXIS.JitterInterval, 0.01, 1, 0.05) then
        jitterTimer = 0
        jitterSide  = -jitterSide
        local js = math.rad(clampNum(AXIS.JitterStrength, 0, 180, 45))
        antiAimJitter = (math.random() * 2 - 1) * js
        randomYaw   = math.random() * TAU
        randomPitch = math.rad(math.random(-89, 89))
        randomRoll  = math.random() * TAU
        local s = clampNum(AXIS.PositionJitterStrength, 0, 5, 1)
        posJitter = Vector3.new((math.random() * 2 - 1) * s, (math.random() * 2 - 1) * s * 0.5, (math.random() * 2 - 1) * s)
    end
end

local function getAntiAimAngles()
    local js = math.rad(clampNum(AXIS.JitterStrength, 0, 180, 45))
    local yaw = math.rad(AXIS.YawAngle or 0)
    local mode = AXIS.YawMode or "Spin"
    if     mode == "Spin"        then yaw += antiAimAngle
    elseif mode == "Jitter"      then yaw += jitterSide * js
    elseif mode == "Random"      then yaw += randomYaw
    elseif mode == "Backwards"   then yaw += math.pi
    elseif mode == "Sideways"    then yaw += math.pi / 2
    elseif mode == "Spin+Jitter" then yaw += antiAimAngle + jitterSide * js
    end
    if AXIS.JitterEnabled then yaw += antiAimJitter end

    local pitch = 0
    local pm = AXIS.PitchMode or "None"
    if     pm == "Up"     then pitch = math.rad(89)
    elseif pm == "Down"   then pitch = math.rad(-89)
    elseif pm == "Random" then pitch = randomPitch
    elseif pm == "Custom" then pitch = math.rad(clampNum(AXIS.PitchAngle, -89, 89, 0))
    end

    local roll = 0
    local rm = AXIS.RollMode or "None"
    if     rm == "Upside Down" then roll = math.pi
    elseif rm == "Random"      then roll = randomRoll
    end
    return pitch, yaw, roll
end

-- baseCF: anti-aim uygulanmamis temel CFrame (konum + bakis)
local function applyAntiAim(hrp, dt, baseCF)
    baseCF = baseCF or CFrame.new(hrp.Position)
    if not AXIS.AntiAimEnabled then hrp.CFrame = baseCF return end
    antiAimTick(dt)
    local pitch, yaw, roll = getAntiAimAngles()
    local cf = baseCF * CFrame.Angles(0, yaw, 0) * CFrame.Angles(pitch, 0, roll)
    if AXIS.PositionJitterEnabled then cf = cf + posJitter end
    hrp.CFrame = cf
end

        if not AXIS.VoidSpamEnabled then

        local interval = voidSpamInterval()   -- 0.005 - 1 sn
        if acc < interval then continue end
        acc %= interval
        for _ = 1, math.clamp(AXIS.VoidSpamBurst or 5, 1, 20) do
            if not hrp.Parent then break end
            goVoid(hrp)
        end
    end
end)

local function waitInPlace(hrp, dt)
    setGameState("WAIT")
    waitAnchor = waitAnchor or hrp.Position
    hrp.AssemblyLinearVelocity  = Vector3.zero
    hrp.AssemblyAngularVelocity = Vector3.zero

end

-- RAGEBOT DONUS NOKTASI: acildigi andaki konum kaydedilir, kapatilinca aninda oraya donulur
local Rage = { was = false, home = nil }

local function returnHome(hrp, hum)
    local home = Rage.home
    Rage.home = nil
    if not home or not hrp or not hrp.Parent then return end
    hrp.AssemblyLinearVelocity  = Vector3.zero
    hrp.AssemblyAngularVelocity = Vector3.zero
    hrp.CFrame = home
    if hum then pcall(function() hum:ChangeState(Enum.HumanoidStateType.GettingUp) end) end
    log("Ragebot kapandi -> eski konuma donuldu")
end

local function step(dt)
    local char = lp.Character
    local hrp  = char and char:FindFirstChild("HumanoidRootPart")
    local hum  = char and char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum or hum.Health <= 0 then return end

    if AXIS.Enabled ~= Rage.was then
        Rage.was = AXIS.Enabled
        if AXIS.Enabled then
            Rage.home = hrp.CFrame          -- acildigi yer
            setGameState("INIT_VOID")
        else
            setSilentTarget(nil)
            RageShoot.part, RageShoot.player = nil, nil
            setGameState("IDLE")
            returnHome(hrp, hum)            -- kapatinca aninda geri don
            return
        end
    end
    if not AXIS.Enabled then setSilentTarget(nil) return end
    -- Void spam acikken konumu spam dongusu yonetir; kapaninca bu makine aynen devam eder
    if AXIS.VoidSpamEnabled then setSilentTarget(nil) return end

    local now = tick()

    -- 1) INIT_VOID
    if GameState == "INIT_VOID" then
        goVoid(hrp)
        if now - StateStart >= hideTime() then setGameState("HUNT") end
        return
    end

    -- 2) RELOAD / MERMI YOK / SILAH DEGISIYOR -> void'de bekle; mermi dolunca saldiriya don
    if isReloading() or not hasAmmo() or isEquipping() then
        setGameState("RELOAD_VOID")
        goVoid(hrp)
        return
    end
    if GameState == "RELOAD_VOID" then
        if now - voidEntryTime < getVoidDelay() then goVoid(hrp) return end
        setGameState("HUNT")
    end

    -- 3) DUSMAN ARA
    local enemy = getClosestEnemy()
    local eChar = enemy and enemy.Character
    local eHrp  = eChar and eChar:FindFirstChild("HumanoidRootPart")
    local aimPart = getAimPart(eChar)

    if not (enemy and eHrp and aimPart) then
        if AXIS.VoidLockEnabled and hasAnyLiveEnemy() then
            waitInPlace(hrp, dt)          -- VOID LOCK: canli dusman varken void'e gitme
            return
        end
        setGameState("VOID_SPAM")
        goVoid(hrp)
        return
    end

    setGameState("HUNT")

    -- 4) HUNT
    local targetPos = getTeleportPosition(eHrp, enemy, eChar)
    if not targetPos then return end

    if (hrp.Position - targetPos).Magnitude > (AXIS.TeleportConfirmTol or 2) then
        if serverSafeTeleport(hrp, targetPos, eHrp.Position) then
            stats.Teleports += 1
            teleportDone, teleportAt = true, now
            -- TeleportToShootDelay 0 ise ayni karede ates et (bir kare kazanc)
            if (AXIS.TeleportToShootDelay or 0) > 0 then return end
        end
    elseif not teleportDone then
        teleportDone, teleportAt = true, now
    end

    if AXIS.RequireTeleport and not teleportDone then return end
    if now - teleportAt < (AXIS.TeleportToShootDelay or 0) then return end

    applyAntiAim(hrp, dt, CFrame.new(targetPos, eHrp.Position))

    local cam = Workspace.CurrentCamera   -- kamera degisirse eski referans kullanilmaz
    if cam then pcall(function() cam.CFrame = CFrame.new(cam.CFrame.Position, aimPart.Position) end) end

    -- Isinlandiktan sonra RAPID FIRE: hedef auto fire'a verilir (oyunun kendi silahi,
    -- silent hook mermiyi kafaya cevirir). Istege bagli ek remote atislar.
    RageShoot.part, RageShoot.player, RageShoot.at = aimPart, enemy, now
    if AXIS.RageRemoteFire and now - lastShot >= (AXIS.ShootInterval or 0) then
        lastShot = now
        fireAt(aimPart)
    end
end

local function isKatanaDeflecting(item)
    if type(item) ~= "table" or item.Name ~= "Katana" then return false end
    local now = tick()
    local ac = rawget(item, "_attack_cooldown")
    local dc = rawget(item, "_deflect_cooldown")
    local dh = rawget(item, "_deflect_hash")
    return type(ac) == "number" and ac > now and type(dc) == "number" and dc > now

local function inventoryItemName(item)
    local t = typeof(item)
    if t == "string" then return item ~= "" and item or nil end
    if t == "Instance" then return item.Name end
    if t ~= "table" then return nil end
    if type(item.Name) == "string" and item.Name ~= "" then return item.Name end
    local info = item.Info
    if type(info) == "table" and type(info.Name) == "string" then return info.Name end
    return nil
end

local function fighterLoadoutNames(fighter)
    local names = {}
    local inv = fighter and (fighter.Loadout or fighter.Slots or fighter.Items or fighter.EquippedItems)
    if type(inv) == "table" then
        for _, it in pairs(inv) do
            local n = inventoryItemName(it)
            if n then names[n] = true end
        end
    end
    local last = fighter and fighter.Data and fighter.Data.LastPickedWeapons
    if type(last) == "table" then for _, n in ipairs(last) do names[n] = true end end
    return names
end

local function isDeflectBlocked(plr, localItem)
    if not AXIS.CheckKatanaDeflect then return false end
    if localItem and DEFLECT_BYPASS_ITEMS[localItem.Name] then return false end
    local f = getFighter(plr)
    return isKatanaDeflecting(f and f.EquippedItem)
end

local function isShieldBlocked(plr, char, localItem, observerPos)
    if not AXIS.CheckRiotShield then return false end
    if localItem and SHIELD_PIERCE_ITEMS[localItem.Name] then return false end
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local f = getFighter(plr)
    if not root or not f or typeof(observerPos) ~= "Vector3" then return false end
    local equipped = inventoryItemName(f.EquippedItem)
    local hasShield = fighterLoadoutNames(f)["Riot Shield"] == true
    local toLocal = observerPos - root.Position
    toLocal = toLocal.Magnitude > 0.001 and toLocal.Unit or root.CFrame.LookVector
    local dot = root.CFrame.LookVector:Dot(toLocal)
    if equipped == "Riot Shield" and dot >= 0 then return true end              -- kalkan onde
    if hasShield and equipped and equipped ~= "Riot Shield" and dot <= 0 then return true end  -- kalkan sirtta
    return false
end

local function canAimWithItem(item)
    if type(item) ~= "table" then return false end
    local name = item.Name
    if AIMBOT_HARD_BLOCK[name] then return false end
    local ignore = AXIS.IgnoreAimbotWeapons
    if type(ignore) == "table" and (ignore[name] == true or table.find(ignore, name)) then return false end
    local info = item.Info
    if type(info) == "table" and type(item.StartShooting) == "function" then
        if type(info.ShootCooldown) == "number" or type(info.ShootBurstCooldown) == "number"
           or type(info.ShootBurst) == "number" or type(info.QuickShotCooldown) == "number"
           or type(info.ChargeReleaseCooldown) == "number" or type(info.BladeCooldown) == "number" then
            return true
        end
        if type(info.AttackCooldown) == "number" and (info.Class == "Melee" or info.Type == "Melee") then return true end
    end
    return name == "Daggers" or name == "Bow" or name == "Slingshot"
end

local RANDOM_PARTS = { "HitboxHead", "Head", "UpperTorso", "HitboxBody", "HumanoidRootPart" }
local randomPartChoice = setmetatable({}, { __mode = "k" })

local function resolveAimbotPart(char)
    local mode = AXIS.AimbotAimPart or "Auto"
    if mode == "Random" then
        local pick = randomPartChoice[char]
        if not pick or not char:FindFirstChild(pick) then
            local avail = {}
            for _, n in ipairs(RANDOM_PARTS) do if char:FindFirstChild(n) then table.insert(avail, n) end end
            pick = avail[math.random(1, math.max(#avail, 1))]
            randomPartChoice[char] = pick
        end
        return pick and char:FindFirstChild(pick)
    end
    local head = char:FindFirstChild("HitboxHead") or char:FindFirstChild("Head")
    if mode == "Head" then return head end
    return head or char:FindFirstChild("HitboxBody") or char:FindFirstChild("HumanoidRootPart")
end

local visParams = RaycastParams.new()
visParams.FilterType = Enum.RaycastFilterType.Exclude
local function isPartVisible(part, char)
    local cam = Workspace.CurrentCamera
    if not cam or not part then return false end
    visParams.FilterDescendantsInstances = { lp.Character, cam }
    local origin = cam.CFrame.Position
    local res = Workspace:Raycast(origin, part.Position - origin, visParams)
    return res == nil or res.Instance:IsDescendantOf(char)
end

local function isValidAimTarget(plr, localItem)
    if plr == lp or isTeammate(plr) then return false end
    local char = plr.Character
    if not char or not aliveHumanoid(char) or hasProtection(char) then return false end
    if isDeflectBlocked(plr, localItem) then return false end
    local myHrp = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
    if isShieldBlocked(plr, char, localItem, myHrp and myHrp.Position) then return false end
    return true
end

local function getAimbotTarget(requireVisible, onScreenOnly, ignoreFov)
    local cam = Workspace.CurrentCamera
    if not cam then return nil end
    local item = equippedItem()
    local mouse = UserInputService:GetMouseLocation()
    local myPos = cam.CFrame.Position
    local best, bestScore = nil, math.huge
    for _, plr in ipairs(Players:GetPlayers()) do
        if isValidAimTarget(plr, item) then
            local char = plr.Character
            local part = resolveAimbotPart(char)
            if part and (not requireVisible or isPartVisible(part, char)) then
                local sp, onScreen = cam:WorldToViewportPoint(part.Position)
                onScreen = onScreen and sp.Z > 0
                local score
                if onScreenOnly and not onScreen then
                    score = nil
                elseif AXIS.IgnoreFOV or ignoreFov then
                    score = (part.Position - myPos).Magnitude
                elseif onScreen then
                    local d = (Vector2.new(sp.X, sp.Y) - mouse).Magnitude
                    if d <= (AXIS.FOVRadius or 282) then score = d end
                end
                if score and score < bestScore then
                    best, bestScore = { player = plr, char = char, part = part, pos = part.Position }, score
                end
            end
        end
    end
    return best
end

local function wrapModuleFn(mod, name, makeWrapper)
    if type(mod) ~= "table" then return false end
    HOOKS[mod] = HOOKS[mod] or {}
    local original = HOOKS[mod][name] or rawget(mod, name)
    if type(original) ~= "function" then return false end
    HOOKS[mod][name] = original
    mod[name] = makeWrapper(original)
    return true
end
local function restoreAllHooks()
    for mod, fns in pairs(HOOKS) do
        for name, orig in pairs(fns) do pcall(function() mod[name] = orig end) end
    end
    table.clear(HOOKS)
end

local function attackSpeedScale()
    if not AXIS.ModAttackSpeed then return 1 end
    return 1 / (1 + clampNum(AXIS.ModAttackSpeedBoost, 0, 85, 85) / 100)
end

local function callWithScaledInfo(info, fields, scale, minValue, fn, ...)
    local saved = {}
    if type(info) == "table" and scale < 0.999 then
        for _, f in ipairs(fields) do
            local v = info[f]
            if type(v) == "number" then saved[f] = v info[f] = math.max(minValue, v * scale) end
        end
    end
    local res = table.pack(pcall(fn, ...))
    for f, v in pairs(saved) do info[f] = v end
    if not res[1] then
        stats.Errors += 1
        if AXIS.ShowDebug then warn("[axis] mod: " .. tostring(res[2])) end
        return table.pack()
    end
    return table.pack(table.unpack(res, 2, res.n))
end

local function shootItem(self)
    if type(self) == "table" and type(self.Name) == "string" and self.Name ~= "" then return self end
    local it = equippedItem()

end

local function resolveReloadKey(item, reloadEnum)

    local a = itemAmmo(item)
    if type(a) == "number" and a <= 0 and item.Info and item.Info.HasEmptyReload then return "EmptyReload" end
    return "Reload"
end

local function installGunHooks()
    local gun = playerModule("ItemTypes", "Gun")
    if not gun then return false end
    wrapModuleFn(gun, "StartShooting", function(orig)
        return function(self, ...)
            local item = shootItem(self)
            if not item then return nil end
            local r = callWithScaledInfo(item.Info, { "ShootCooldown", "ShootBurstCooldown", "QuickShotCooldown" },
                attackSpeedScale(), 0.001, orig, item, ...)
            -- Harion No Spread: StartShooting sonucunun 4. degeri no-spread bayragi
            if AXIS.HarionNoSpread and typeof(r[3]) == "table" then
                r[4] = true
                if (r.n or 0) < 4 then r.n = 4 end
            end
            return table.unpack(r, 1, r.n)
        end
    end)
    wrapModuleFn(gun, "StartReloading", function(orig)
        return function(self, forceReload, reloadEnum, animationEnum, segmentedCount)
            local boost = clampNum(AXIS.ModReloadBoost, 0, 20, 20)
            if not AXIS.ModFasterReload or boost <= 0 or type(self) ~= "table" or type(self.Info) ~= "table" then
                return orig(self, forceReload, reloadEnum, animationEnum, segmentedCount)
            end
            local key = resolveReloadKey(self, reloadEnum)
            local info = self.Info
            local fields
            if info.ReloadType == "Regular" then
                fields = { key .. "Length", key .. "ActionTimestamp" }
            elseif info.ReloadType == "Segmented" then
                fields = { key .. "StartLength", key .. "StartActionTimestamp", key .. "SegmentLength",
                    key .. "SegmentActionTimestamp", key .. "FinishLength", key .. "FinishActionTimestamp" }
            else
                return orig(self, forceReload, reloadEnum, animationEnum, segmentedCount)
            end
            local r = callWithScaledInfo(info, fields, 1 - boost / 100, 0.05, orig, self, forceReload, reloadEnum, animationEnum, segmentedCount)
            return table.unpack(r, 1, r.n)
        end
    end)
    wrapModuleFn(gun, "_Recoil", function(orig)
        return function(self, multiplier)
            if AXIS.ModNoRecoil and type(multiplier) == "number" then
                local m = multiplier * (1 - clampNum(AXIS.ModRecoilReduction, 0, 100, 100) / 100)
                if m <= 0.001 then return end
                return orig(self, m)
            end
            return orig(self, multiplier)
        end
    end)
    wrapModuleFn(gun, "GetAimSpeed", function(orig)
        return function(self)
            local s = orig(self)
            if AXIS.ModFasterADS and type(s) == "number" then
                return math.max(1, s * (1 + clampNum(AXIS.ModADSBoost, 0, 85, 85) / 100))
            end
            return s
        end
    end)
    wrapModuleFn(gun, "Equip", function(orig)
        return function(self, ...)
            local r = table.pack(orig(self, ...))
            if AXIS.ModFasterEquip and type(self) == "table" then
                local scale = 1 / (1 + clampNum(AXIS.ModEquipBoost, 0, 80, 80) / 100)
                local now = tick()
                local remaining = type(self._equip_cooldown) == "number" and math.max(self._equip_cooldown - now, 0) or 0
                self._equip_cooldown = now + remaining * scale
                local hash = self._equip_hash
                task.delay(remaining * scale, function()
                    if self and self._equip_hash == hash then stopEquipAnimations(self) end
                end)
            end
            return table.unpack(r, 1, r.n)
        end
    end)
    return true
end

-- Kalici Info degisiklikleri (Automatic Weapon / Scythe Dash), kapaninca orijinale doner
local ItemInfoOriginals = setmetatable({}, { __mode = "k" })
local function applyPersistentMods()
    local f = localFighter()
    local items = type(f) == "table" and rawget(f, "Items") or nil
    if type(items) ~= "table" then return end
    for _, item in pairs(items) do
        local info = type(item) == "table" and item.Info or nil
        if type(info) == "table" then
            local orig = ItemInfoOriginals[info]
            if not orig then
                local ise = rawget(info, "InputSpammingEnabled")
                orig = {
                    HasISE = ise ~= nil, ISE = type(ise) == "table" and table.clone(ise) or ise,
                    HasDash = rawget(info, "DashCooldown") ~= nil, Dash = rawget(info, "DashCooldown"),
                }
                ItemInfoOriginals[info] = orig
            end
            if AXIS.ModAutomatic and type(orig.ISE) == "table" then
                local auto = table.clone(orig.ISE)
                auto.StartShooting, auto.StartReloading = 0, 0
                rawset(info, "InputSpammingEnabled", auto)
            else
                rawset(info, "InputSpammingEnabled", orig.HasISE and (type(orig.ISE) == "table" and table.clone(orig.ISE) or orig.ISE) or nil)
            end
            if AXIS.ModScytheDash and type(orig.Dash) == "number" then
                rawset(info, "DashCooldown", orig.Dash * (1 - clampNum(AXIS.ModDashReduction, 0, 100, 75) / 100))
            else
                rawset(info, "DashCooldown", orig.HasDash and orig.Dash or nil)
            end
        end
    end
end
local function restorePersistentMods()
    for info, orig in pairs(ItemInfoOriginals) do
        pcall(function()
            rawset(info, "InputSpammingEnabled", orig.HasISE and orig.ISE or nil)
            rawset(info, "DashCooldown", orig.HasDash and orig.Dash or nil)
        end)
    end
end

-- Input hook (No Spread / Grenade Fuse): ClientItem.Input'un ReplicatedStorage upvalue'su
-- sahte bir agaca yonlendirilir, UseItem paketleri gonderilmeden once duzenlenir.
local InputHook = { restore = nil, trajectory = nil }

local function getLocalItemById(objectId)
    local f = localFighter()
    local items = type(f) == "table" and rawget(f, "Items") or nil
    if type(items) ~= "table" then return nil end
    for _, item in next, items do
        local d = type(item) == "table" and rawget(item, "Data") or nil
        if type(d) == "table" and rawget(d, "ObjectID") == objectId then return item end
    end
    return nil
end

local function computeImpactTime(tv)
    local lastArgs = rawget(tv, "_last_args") or {}
    local step = lastArgs[4] or 0.05
    local cap = lastArgs[6] or math.huge
    local sphere = rawget(tv, "_impact_sphere")
    local segments = rawget(tv, "_segments")
    if not sphere or type(segments) ~= "table" then return nil end
    local impactCF = sphere.CFrame
    for key, seg in next, segments do
        if seg.CFrame == impactCF then return math.min(cap, (key - 2) * step) end
    end
    return nil
end

local function handleInputPacket(packet)
    if type(packet.args) ~= "table" then return end
    local t = packet.type
    if AXIS.ModNoSpread and t == "StartShooting" then
        local item = getLocalItemById(packet.objectId)
        local info = item and rawget(item, "Info")
        if type(info) == "table" and rawget(info, "Type") == "Gun" then packet.args["\2"] = true end
    end
    if AXIS.ModGrenadeFuse then
        local isStart = t == "StartShooting" or t == "StartAiming"
        local isFinish = t == "FinishShooting" or t == "FinishAiming"
        if not (isStart or isFinish) then return end
        local item = getLocalItemById(packet.objectId)
        local info = item and rawget(item, "Info")
        if type(info) ~= "table" or not rawget(info, "CanCook") then return end
        if isStart then
            InputHook.trajectory = rawget(item, "_trajectory_visual")
            if AXIS.ModRemoveFuse then rawset(item, "_cook_detonate_delay", nil) end
            return
        end
        if AXIS.ModExplodeOn == "Throw" then
            packet.args["\3"] = 0
        else
            local tv = InputHook.trajectory
            packet.args["\3"] = tv and computeImpactTime(tv) or nil
        end
        InputHook.trajectory = nil
    end
end

local function installInputHook()
    if InputHook.restore then return true end
    if not (debug and type(debug.getupvalues) == "function" and type(debug.setupvalue) == "function"
            and type(clonefunction) == "function") then return false end
    local clientItem = playerModule("ClientReplicatedClasses", "ClientFighter", "ClientItem")
    local fromEnum = EnumLibrary and rawget(EnumLibrary, "_from_enum")
    local inputMethod = clientItem and rawget(clientItem, "Input")
    if type(inputMethod) ~= "function" or type(fromEnum) ~= "table" then return false end

    local fireNative = clonefunction(Instance.new("RemoteEvent").FireServer)
    local remote = (type(cloneref) == "function" and cloneref(UseItemRemote)) or UseItemRemote
    local function fireServer(_, objectId, encodedType, args)
        local packet = { block = false, objectId = objectId, type = rawget(fromEnum, encodedType), args = args }
        safeCall("InputHook", handleInputPacket, packet)
        if packet.block then return end
        return fireNative(remote, objectId, encodedType, packet.args, nil)
    end
    local tree = { Replication = { Fighter = { UseItem = { FireServer = fireServer } } } }
    local dummy = newproxy(true)                       -- kendi metatablosu olan userdata (global etki yok)
    getmetatable(dummy).__index = function() return tree end

    for idx, v in pairs(debug.getupvalues(inputMethod)) do
        if v == ReplicatedStorage then
            debug.setupvalue(inputMethod, idx, dummy)
            InputHook.restore = { method = inputMethod, index = idx, original = v }
            break
        end
    end
    if not InputHook.restore then
        for _, bundle in pairs(debug.getupvalues(inputMethod)) do
            if type(bundle) == "table" then
                for k, v in pairs(bundle) do
                    if v == ReplicatedStorage then
                        bundle[k] = dummy
                        InputHook.restore = { bundle = bundle, key = k, original = v }
                        break
                    end
                end
            end
            if InputHook.restore then break end
        end
    end
    return InputHook.restore ~= nil
end
local function restoreInputHook()
    local r = InputHook.restore
    if not r then return end
    InputHook.restore = nil
    pcall(function()
        if r.bundle then r.bundle[r.key] = r.original else debug.setupvalue(r.method, r.index, r.original) end
    end)
end

-- Hook'lar SADECE ilgili bir ozellik acilinca kurulur (KiciaHook EnsureHooks gibi).
-- Execute aninda oyun modullerine dokunulmaz -> execute'ta donma / tespit riski yok.
local function modsNeeded()
    return AXIS.ModAttackSpeed or AXIS.ModFasterReload or AXIS.ModNoRecoil or AXIS.ModFasterADS
        or AXIS.ModFasterEquip or AXIS.ModRemoveFuse or AXIS.HarionNoSpread
end
local persistTouched = false
loop(1, function()
    if modsNeeded() then
        if not ModsInstalled.Gun then ModsInstalled.Gun = installGunHooks() end
        if not ModsInstalled.Melee then ModsInstalled.Melee = installMeleeHooks() end
        if not ModsInstalled.Gunblade then ModsInstalled.Gunblade = installGunbladeHooks() end
        if not ModsInstalled.Throwable then ModsInstalled.Throwable = installThrowableHooks() end
    end
    if (AXIS.ModNoSpread or AXIS.ModGrenadeFuse) and not InputHook.restore then
        if not installInputHook() and AXIS.ShowDebug and not InputHook.warned then
            InputHook.warned = true
            warn("[axis] No Spread / Grenade Fuse: executor debug.getupvalues/setupvalue desteklemiyor")
        end
    end
    if AXIS.ModAutomatic or AXIS.ModScytheDash then
        persistTouched = true
        applyPersistentMods()
    elseif persistTouched then
        persistTouched = false
        restorePersistentMods()
    end
end)

local Flick = { active = false, lastEnd = 0 }

local function bezierDir(a, m, b, t)
    local u = 1 - t
    local d = a * (u * u) + m * (2 * u * t) + b * (t * t)
    return d.Magnitude > 0 and d.Unit or b
end

local function updateFlickbot(cam, now)
    if not AXIS.FlickbotEnabled then Flick.active = false return end
    if Flick.active then
        local tgt = Flick.target
        local part = tgt and tgt.part
        if not part or not part.Parent then Flick.active = false Flick.lastEnd = now return end
        local camPos = cam.CFrame.Position
        local toDir = (part.Position - camPos).Unit
        local x = math.clamp((now - Flick.start) / Flick.duration, 0, 1)
        local t = 1 - (1 - x) ^ 3                                   -- ease-out
        local dir = bezierDir(Flick.fromDir, Flick.midDir, toDir, t)
        local noise = (1 - t) * Flick.noise
        dir = (dir + Vector3.new((math.random() - 0.5) * noise, (math.random() - 0.5) * noise, 0)).Unit
        cam.CFrame = CFrame.lookAt(camPos, camPos + dir)
        if x >= 1 then
            Flick.active = false
            Flick.lastEnd = now
            if AXIS.FlickShoot and canAimWithItem(equippedItem()) then
                task.delay(clampNum(AXIS.FlickShotDelay, 0, 250, 0) / 1000, clickOnce)
            end
        end
        return
    end
    if not keyState("AXIS_FlickKey", false) then return end
    if (now - Flick.lastEnd) * 1000 < clampNum(AXIS.FlickCooldown, 0, 2000, 250) then return end
    if not canAimWithItem(equippedItem()) then return end
    local tgt = getAimbotTarget(false)
    if not tgt then return end
    local cf = cam.CFrame
    local fromDir = cf.LookVector
    local toDir = (tgt.pos - cf.Position).Unit
    local humanness = clampNum(AXIS.FlickHumanness, 0, 100, 30) / 100
    local curve = clampNum(AXIS.FlickCurvature, 0, 50, 12) / 100
    local side = math.random() < 0.5 and -1 or 1
    local mid = ((fromDir + toDir) / 2 + cf.RightVector * curve * side + cf.UpVector * curve * 0.3 * (math.random() - 0.5))
    Flick.active = true
    Flick.target = tgt
    Flick.start = now
    Flick.duration = clampNum(AXIS.FlickDuration, 30, 400, 110) / 1000 * (1 + (math.random() - 0.5) * humanness * 0.4)
    Flick.fromDir, Flick.midDir = fromDir, mid
    Flick.noise = humanness * 0.02
end

        if p then return p end
        node = node.Parent
    end
    return nil
end

-- ------------------------------------------------------------
-- SILENT AIM: KiciaHook gibi KUSANILAN ESYANIN kendi StartShooting'i sarilir
-- (modul yamasi yetmiyor: esyalar fonksiyonu kendi tablosunda tasiyor)
-- ------------------------------------------------------------
local SilentHook = { item = nil, orig = nil, hadOwn = false, wrapped = nil, depth = 0 }

local function resolveBaseStartShooting(orig)
    -- esyanin fonksiyonu bir modulun orijinaliyse, modulun guncel (mod'lu) surumunu kullan
    for mod, fns in pairs(HOOKS) do
        if fns.StartShooting == orig and type(mod.StartShooting) == "function" then return mod.StartShooting end
    end
    return orig
end

local function unhookSilentItem()
    local it = SilentHook.item
    if it and SilentHook.wrapped and rawget(it, "StartShooting") == SilentHook.wrapped then
        pcall(function() rawset(it, "StartShooting", SilentHook.hadOwn and SilentHook.orig or nil) end)
    end
    SilentHook.item, SilentHook.orig, SilentHook.wrapped = nil, nil, nil
end

local function ensureSilentItemHook(item)
    if SilentHook.item == item then return end
    unhookSilentItem()
    if type(item) ~= "table" or type(item.StartShooting) ~= "function" or not canAimWithItem(item) then return end
    local own = rawget(item, "StartShooting")
    local orig = own or item.StartShooting
    local wrapped
    wrapped = function(self, ...)
        local prime = nil
        if SilentHook.depth == 0 then prime = primeSilentShot(self) end
        SilentHook.depth += 1
        local res = table.pack(pcall(resolveBaseStartShooting(orig), self, ...))
        SilentHook.depth -= 1
        finishSilentShot(prime)
        if not res[1] then
            stats.Errors += 1
            if AXIS.ShowDebug then warn("[axis] silent StartShooting: " .. tostring(res[2])) end
            return nil
        end
        return table.unpack(res, 2, res.n)
    end
    SilentHook.item, SilentHook.orig, SilentHook.hadOwn, SilentHook.wrapped = item, orig, own ~= nil, wrapped
    rawset(item, "StartShooting", wrapped)
end

-- AUTO FIRE: Triggerbot + Silent Rapid Fire + Ragebot ayni anda (KiciaHook TryTriggerbotShot)

local Fire = { target = nil, since = 0, lastShot = 0 }
local SHOT_FIELDS = { "ShootCooldown", "ShootBurstCooldown", "QuickShotCooldown", "ChargeReleaseCooldown", "AttackCooldown", "BladeCooldown" }

local function weaponShotInterval(item)
    local info = type(item) == "table" and item.Info or nil
    local s = 0.05
    if type(info) == "table" then
        for _, f in ipairs(SHOT_FIELDS) do
            if type(info[f]) == "number" then s = math.max(s, info[f]) end
        end
    end
    s = math.max(0.05, s * attackSpeedScale())
    return math.max(s, clampNum(AXIS.SilentRapidInterval, 0.01, 0.2, 0.03))
end

local function itemBusy(item)
    local now = tick()
    local sc = rawget(item, "_shoot_cooldown")
    if type(sc) == "number" and sc > now then return true end
    if itemIsReloading(item) or rawget(item, "_is_charging") == true then return true end
    local a = itemAmmo(item)
    return type(a) == "number" and a <= 0
end

local function cursorTarget(cam, item)
    local m = UserInputService:GetMouseLocation()
    local ray = cam:ViewportPointToRay(m.X, m.Y)
    trigParams.FilterDescendantsInstances = { lp.Character, cam }
    local hit = Workspace:Raycast(ray.Origin, ray.Direction * 1000, trigParams)
    local plr = hit and playerFromPart(hit.Instance)
    if plr and isValidAimTarget(plr, item) then return plr end
    return nil
end

local function updateAutoFire(cam, now)
    local item = equippedItem()
    local rageFresh = AXIS.Enabled and RageShoot.part ~= nil and RageShoot.part.Parent ~= nil and (now - RageShoot.at) < 0.25
    local silentOn = AXIS.AimbotEnabled and AXIS.AimbotMode == "Silent" and aimbotActive()
    local trigOn = AXIS.TriggerbotEnabled and keyState("AXIS_TriggerKey", true)
    local silentAuto = silentOn and AXIS.SilentRapidFire

    if not (rageFresh or trigOn or silentAuto) then Fire.target = nil return end
    if not rageFresh and UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) then Fire.target = nil return end
    if not canAimWithItem(item) then Fire.target = nil return end
    if itemBusy(item) then return end

    -- hedef: ragebot hedefi > silent hedefi (gorunen, ekranda) > imlecin altindaki dusman
    local targetId
    if rageFresh then
        targetId = RageShoot.player
    elseif silentOn then
        local t = getAimbotTarget(true, false, true)   -- 360 derece: kamera arkaya donukken de ates eder
        targetId = t and t.player
    end
    if not targetId and trigOn then targetId = cursorTarget(cam, item) end
    if not targetId then Fire.target = nil return end

    if Fire.target ~= targetId then Fire.target, Fire.since = targetId, now end
    if not rageFresh and trigOn and (now - Fire.since) < clampNum(AXIS.TriggerReaction, 0, 0.1, 0) then return end
    if (now - Fire.lastShot) < weaponShotInterval(item) then return end
    Fire.lastShot = now
    clickOnce()      -- StartShooting -> silent hook mermiyi hedefe cevirir
end

local function aimRenderStep()
    if not Running then return end
    local cam = Workspace.CurrentCamera
    if not cam then return end
    local now = tick()

    if fovCircle then
        fovCircle.Visible = AXIS.AimbotEnabled and AXIS.ShowFOV and not AXIS.IgnoreFOV
        if fovCircle.Visible then
            fovCircle.Position = UserInputService:GetMouseLocation()
            fovCircle.Radius = AXIS.FOVRadius or 282
        end
    end

    -- Silent veya Ragebot aciksa kusanilan esyaya silent hook
    if (AXIS.AimbotEnabled and AXIS.AimbotMode == "Silent") or AXIS.Enabled then
        ensureSilentItemHook(equippedItem())
    elseif SilentHook.item then
        unhookSilentItem()
    end

    -- Camera modu: sadece nisan alirken (sag tik) yumusak takip
    if AXIS.AimbotMode == "Camera" and aimbotActive() and not Flick.active
       and UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)
       and canAimWithItem(equippedItem()) then
        local tgt = getAimbotTarget(false)
        if tgt then
            local alpha = clampNum(AXIS.AimSpeed, 1, 100, 35) / 100
            cam.CFrame = cam.CFrame:Lerp(CFrame.lookAt(cam.CFrame.Position, tgt.pos), alpha)
        end
    end

    safeCall("Flickbot", updateFlickbot, cam, now)
    safeCall("AutoFire", updateAutoFire, cam, now)
end

local RankProfile
local function resolveRankName(elo)
    elo = tonumber(elo)
    if not elo then return "Unranked" end
    if not RankProfile then
        local lib = requirePath(ReplicatedStorage, "Modules", "SeasonLibrary")
        local season = lib and lib.CurrentSeason
        local profile = type(season) == "table" and season.RankProfile or nil
        local pname = type(profile) == "table" and profile.Name or nil
        if lib and type(lib.RankProfiles) == "table" and pname then profile = lib.RankProfiles[pname] or profile end
        if type(profile) == "table" and type(profile.RanksOrder) == "table" and type(profile.Ranks) == "table" then
            RankProfile = { order = profile.RanksOrder, ranks = profile.Ranks }
        else
            return "Unranked"
        end
    end
    local order, ranks = RankProfile.order, RankProfile.ranks
    for i, name in ipairs(order) do
        local r = ranks[name]
        if type(r) == "table" and type(r.RequiredELO) == "number" and elo < r.RequiredELO and r.RequiredELOLeaderboardRanking == nil then
            return order[math.clamp(i - 1, 1, #order)] or "Unranked"
        end
    end
    for i = #order, 1, -1 do
        local r = ranks[order[i]]
        if type(r) == "table" and r.RequiredELOLeaderboardRanking == nil then return order[i] end
    end
    return "Unranked"
end

            if type(info) == "table" and info.CanBeStartedFromClient ~= false and (not rel or rel <= serverTime) then add(k) end
        end

    local rot = requirePath(ReplicatedStorage, "Modules", "RotatingQueueLibrary")

        if ok and type(cur) == "table" and type(cur.QueueNames) == "table" then for _, q in ipairs(cur.QueueNames) do add(q) end end

local function autoQueueJoin()
    if not AXIS.AutoQueue or not canFireQueue() then return false end
    if AutoQueue.believed == "queued" or AutoQueue.believed == "matched" then return false end
    if tick() - AutoQueue.lastFire < 2 then return false end
    AutoQueue.lastFire = tick()
    resolveQueueModes()
    local key = AutoQueue.map[AXIS.AutoQueueMode] or AXIS.AutoQueueMode
    if type(key) ~= "string" or key == "" then return false end

    local arcade = AutoQueue.arcade[key]
    if arcade then
        local placeId = tonumber(arcade.PlaceID)
        if placeId and game.PlaceId == placeId then return true end
        local ctl = playerController("ArcadeController")
        local ok, res
        if ctl and type(ctl.ToArcadeServer) == "function" then
            ok, res = pcall(ctl.ToArcadeServer, ctl, key)
        else
            local rem = queueRemote("Arcade", "TeleportToArcadeServer")
            if rem then ok, res = pcall(function() return rem:InvokeServer(key) end) end
        end
        if ok and not (type(res) == "string" and res ~= "") then AutoQueue.believed = "matched" return true end
        return false
    end

    local ctl = playerController("MatchmakingController")
    if ctl and type(ctl.QueueInto) == "function" then
        local ok, res = pcall(function() return ctl:QueueInto(key) end)
        if ok and not (type(res) == "string" and #res > 0) then
            AutoQueue.believed = "queued"
            log("Auto Queue: " .. key)
            return true
        end
    else
        local rem = queueRemote("Matchmaking", "JoinQueue")
        if rem and rem:IsA("RemoteFunction") then
            local ok, res = pcall(function() return rem:InvokeServer(key) end)
            if ok and not (type(res) == "string" and #res > 0) then AutoQueue.believed = "queued" return true end
        end
    end
    return false
end

        if rem and rem:IsA("RemoteEvent") then pcall(function() rem:FireServer() end) end
    end
    AutoQueue.believed = "idle"
end

    return char:FindFirstChild("InvincibilityParticles", true) ~= nil
end

local function isKatana(plr) return scanWeapon(plr):find("katana", 1, true) ~= nil end
local function isRiotShield(plr)
    local w = scanWeapon(plr)
    return w:find("riot", 1, true) ~= nil or w:find("shield", 1, true) ~= nil
end
local function sameMatch(plr) return plr:GetAttribute("EnvironmentID") == lp:GetAttribute("EnvironmentID") end

local function isNearOtherMatch(pos, ignore)
    if typeof(pos) ~= "Vector3" then return false end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= lp and plr ~= ignore and not sameMatch(plr) then
            local r = getRoot(plr.Character)
            if r and (r.Position - pos).Magnitude <= 1000 then return true end
        end
    end
    return false
end

local function shouldSkip(plr)
    if plr == lp or playerIsDead(plr) then return true end
    if not sameMatch(plr) or isInvincible(plr) then return true end
    local r = getRoot(plr.Character)
    if r and isNearOtherMatch(r.Position, plr) then return true end
    return r and r:FindFirstChild("TeammateLabel") ~= nil
end

local function getBestTarget()
    local root = getRoot(lp.Character)
    if not root then return nil end
    local best, bestV = nil, math.huge
    local useHP = AXIS.RageTargetMode == "Lowest Health"
    for _, plr in ipairs(Players:GetPlayers()) do
        if not shouldSkip(plr) then
            local char = plr.Character
            local tr = getRoot(char)
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            local v = useHP and hum.Health or (tr.Position - root.Position).Magnitude
            if AXIS.RageAutoPriority then
                if AXIS.RagePriorityVoided and tr.Position.Magnitude > 1000000 then v -= 2000000000 end
                if AXIS.RagePriorityAttackers and scanWeapon(plr) ~= "" then v -= 1000000000 end
            end
            if v < bestV then bestV, best = v, plr end
        end
    end
    return best
end

local function hasValidTarget()
    return state.target ~= nil and not playerIsDead(state.target) and not isInvincible(state.target)
end

local function setStatus()
    local t = hasValidTarget() and state.target or nil
    local voiding = not t or ((AXIS.RageMode == "Void" or AXIS.RageMode == "Orbit") and not state.voidExposed)
    if state.active then setGameState(voiding and "RAGE_VOID" or ("RAGE: " .. t.Name)) end
end

local function shouldShoot()
    if not hasValidTarget() or isKatana(state.target) then return false end
    if AXIS.RageMode == "Void" and not state.voidExposed then return false end
    return true
end

local function equippedSlot()
    local item = equippedItem()
    local s = itemGet(item, "Slot")
    return tonumber(s)
end

local function autoEquipPreferred()
    if not AXIS.RageAutoEquip then return end
    local pref = AXIS.RagePreferredWeapon or "primary"
    local want = (pref == "secondary" and 2) or (pref == "melee" and 3) or 1
    local slot = equippedSlot()
    if slot and slot ~= want and tick() >= (state.ammoActionAt or 0) and tick() >= (state.equipAt or 0) then
        state.equipAt = tick() + 0.5
        pressKey(slotKey[want])
    end
end

local function handleAmmo()
    local fighter = localFighter()
    local item = fighter and fighter.EquippedItem
    if not fighter or not item then return false end
    local ammo = itemGet(item, "Ammo") or 0
    local slot = itemGet(item, "Slot") or 1
    local now = tick()
    local reloading = false
    pcall(function() reloading = fighter:Get("Reloading") == true end)
    if reloading or itemIsReloading(item) then
        state.hideOrbitUntil = math.max(state.hideOrbitUntil or 0, now + 0.25)
        state.ammoActionAt = math.max(state.ammoActionAt or 0, now + 0.1)
        return true
    end
    if type(ammo) ~= "number" or ammo > 0 then return false end
    if now < (state.ammoActionAt or 0) then return true end
    if not AXIS.RageAutoSwap then return true end
    local primary = AXIS.RagePreferredWeapon == "secondary" and 2 or 1
    local secondary = primary == 1 and 2 or 1
    if slot == primary then
        state.ammoActionAt = now + 0.45
        state.hideOrbitUntil = math.max(state.hideOrbitUntil or 0, now + 0.45)
        pressKey(slotKey[secondary])
        return true
    end
    if slot == secondary then
        state.ammoActionAt = now + 0.6
        state.hideOrbitUntil = math.max(state.hideOrbitUntil or 0, now + 0.75)
        pressKey(slotKey[primary])
        task.delay(0.18, function()
            if not state.active then return end
            local i2 = equippedItem()
            if i2 and (itemGet(i2, "Slot") or 1) == primary and (itemGet(i2, "Ammo") or 0) <= 0 then pressKey(Enum.KeyCode.R) end
        end)
        return true
    end
    state.ammoActionAt = now + 0.5
    pressKey(Enum.KeyCode.R)
    return true
end

local function buildCameraData(fromPos, part)
    if not Utility or not part then return nil end
    local look = CFrame.new(fromPos, part.Position)
    local data = {
        [utf8.char(1)] = {
            [utf8.char(0)] = Utility:EncodeCFrame(look),
            [utf8.char(1)] = Utility:EncodeCFrame(look),
            [utf8.char(2)] = part,
            [utf8.char(3)] = Utility:EncodeCFrame(part.CFrame:ToObjectSpace(CFrame.new(part.Position))),
        },
    }
    if AXIS.ModNoSpread or AXIS.HarionNoSpread then data[utf8.char(2)] = true end
    return data
end

local function doFire(part)
    local fighter = localFighter()
    local item = fighter and fighter.EquippedItem
    if not item or not part then return false end
    local cam = Workspace.CurrentCamera
    local fromPos = (AXIS.RageMode == "Teleport" and state.lastTpPos) or (state.csyncCF and state.csyncCF.Position) or (cam and cam.CFrame.Position) or part.Position
    local any = false
    for _ = 1, math.clamp(math.floor(AXIS.RageShootAttempts or 1), 1, 10) do
        local fired = false
        local ammo = itemGet(item, "Ammo") or 0
        if EnumLibrary and Utility and type(ammo) == "number" and ammo > 0 then
            local oid = itemGet(item, "ObjectID")
            local data = buildCameraData(fromPos, part)
            if oid and data then
                fired = pcall(function() UseItemRemote:FireServer(oid, START_SHOOTING, data, nil) end)
            end
        end
        if not fired and type(item.UseItem) == "function" then fired = pcall(function() item:UseItem() end) end
        any = any or fired
    end
    if any then stats.Shots += 1 end
    return any
end

local function inMatch()
    local now = tick()
    if now - rbInMatchT < 0.25 then return rbInMatch end

-- konumlar: stud (dikey) ve distance (yatay) UI'dan
local function voidShootPos(tr)
    return tr.Position - tr.CFrame.LookVector * dist() + Vector3.new(0, stud(), 0)
end
local function shieldPos(tr)
    return tr.Position - tr.CFrame.LookVector * (dist() + 1.5) + Vector3.new(0, stud(), 0)
end
local function pickOffset(tr, head)
    local look, right = tr.CFrame.LookVector, tr.CFrame.RightVector
    local dirs = { -look, -right, right }
    return head.Position + dirs[math.random(1, #dirs)] * math.max(dist(), 1.25) + Vector3.new(0, stud(), 0)
end

local function setCsync(cf, pos, dt)
    local old = state.lastFakePos
    state.csyncCF = cf
    state.csyncLV = (old and dt and dt > 0) and (pos - old) / dt or Vector3.zero
    state.csyncAV = Vector3.zero
    state.lastFakePos = pos
end
local function clearCsync() state.csyncCF, state.csyncLV, state.csyncAV, state.lastFakePos = nil, nil, nil, nil end
local function setVoidCsync(cf)
    state.csyncCF, state.csyncLV, state.csyncAV = cf, Vector3.zero, Vector3.zero
    state.lastFakePos = cf and cf.Position or nil
end

local function restoreLocalRoot(root)
    if not root or not state.csyncLocalCF then return false end
    local live = root.AssemblyLinearVelocity
    root.CFrame = state.csyncLocalCF
    if state.csyncLocalLV then root.AssemblyLinearVelocity = Vector3.new(state.csyncLocalLV.X, live.Y, state.csyncLocalLV.Z) end
    if state.csyncLocalAV then root.AssemblyAngularVelocity = state.csyncLocalAV end
    return true
end

local function startCsync()
    if state.csyncHbConn then return end
    state.csyncHbConn = RunService.Heartbeat:Connect(function()
        local root = getRoot(lp.Character)
        if not root then return end
        if state.csyncWroteFake and state.csyncLocalCF then restoreLocalRoot(root) end
        state.csyncLocalCF = root.CFrame
        state.csyncLocalLV = root.AssemblyLinearVelocity
        state.csyncLocalAV = root.AssemblyAngularVelocity
        if state.csyncCF then
            root.CFrame = state.csyncCF
            local fv = state.csyncLV or state.csyncLocalLV
            root.AssemblyLinearVelocity = Vector3.new(fv.X, state.csyncLocalLV.Y, fv.Z)
            root.AssemblyAngularVelocity = state.csyncAV or state.csyncLocalAV
            state.csyncWroteFake = true
        else
            state.csyncWroteFake = false
        end
    end)
    RunService:BindToRenderStep("AXIS_RagebotCsync", Enum.RenderPriority.Camera.Value - 1, function()
        local root = getRoot(lp.Character)
        if root and state.csyncLocalCF and state.csyncWroteFake and restoreLocalRoot(root) then
            state.csyncWroteFake = false
        end
    end)
end

local function stopCsync()
    if state.csyncHbConn then state.csyncHbConn:Disconnect() state.csyncHbConn = nil end
    pcall(function() RunService:UnbindFromRenderStep("AXIS_RagebotCsync") end)
    restoreLocalRoot(getRoot(lp.Character))
    clearCsync()
    state.csyncLocalCF, state.csyncLocalLV, state.csyncLocalAV, state.csyncWroteFake = nil, nil, nil, false
end

local function voidRand()
    local n
    repeat n = math.random(-2147483646, 2147483646) until n < -1147483646 or n > 1147483646
    return n
end
local function voidRandCF() return CFrame.new(voidRand(), voidRand(), voidRand()) * CFrame.Angles(math.pi, math.pi, math.pi) end

local function enterVoidState()
    state.voidTargetCF, state.voidExposed = nil, false
    if not state.active then clearCsync() return end
    if AXIS.RageVoidSpam then setVoidCsync(voidRandCF()) else clearCsync() end
    setStatus()
end

local function enableVoidCsync()
    if state.voidHbConn then return end
    startCsync()
    state.voidHbConn = RunService.Heartbeat:Connect(function()
        if state.voidTargetCF then setVoidCsync(state.voidTargetCF)
        elseif AXIS.RageVoidSpam then setVoidCsync(voidRandCF())
        else clearCsync() end
    end)
end
local function disableVoidCsync()
    if state.voidHbConn then state.voidHbConn:Disconnect() state.voidHbConn = nil end
    state.voidTargetCF, state.voidThread, state.voidExposed = nil, nil, false
end

local function startVoidLoop(myGen)
    if state.voidThread then return end
    enableVoidCsync()
    local th
    th = task.spawn(function()
        while state.active and rbGen == myGen and not state.suspended do
            if not inMatch() or not hasValidTarget() or isKatana(state.target) then
                enterVoidState()
                task.wait(0.1)
                continue
            end
            enterVoidState()
            local hide = hideTime()
            if hide > 0 then task.wait(hide) end
            if not state.active or rbGen ~= myGen or state.suspended or not inMatch() then break end
            local target = state.target
            if hasValidTarget() and not isKatana(target) then
                local tc = target.Character
                local tr = getRoot(tc)
                local head = tc and (tc:FindFirstChild("Head") or tr)
                if tr and head then
                    local shootPos = isRiotShield(target) and shieldPos(tr) or voidShootPos(tr)
                    if isNearOtherMatch(shootPos, target) then
                        enterVoidState()
                        task.wait(0.1)
                        continue
                    end
                    local shootCF = CFrame.new(shootPos, head.Position)
                    state.voidExposed, state.voidTargetCF = true, shootCF
                    setVoidCsync(shootCF)
                    setStatus()
                    local atk = clampNum(AXIS.RageAttackTime, 0, 1, 0.03)
                    if atk > 0 then task.wait(atk) end
                    if hasValidTarget() and not isKatana(target) then doFire(head) end
                    task.wait(0.05)
                    enterVoidState()
                else
                    task.wait(0.05)
                end
            else
                task.wait(0.05)
            end
            if hide <= 0 then task.wait() end   -- her turda en az bir yield (sonsuz dongu/donma yok)
        end
        if state.voidThread == th then state.voidThread = nil end
        if rbGen == myGen and not state.suspended and state.voidThread == nil then disableVoidCsync() end
    end)
    state.voidThread = th
end

local function noclipTick()
    if not AXIS.RageNoclip then return end
    local now = tick()
    if now - state.noclipAt < 0.2 then return end
    state.noclipAt = now
    local char = lp.Character
    if not char then return end
    for _, p in ipairs(char:GetDescendants()) do
        if p:IsA("BasePart") and p.CanCollide then
            if state.collide[p] == nil then state.collide[p] = true end
            p.CanCollide = false
        end
    end
end
local function restoreCollide()
    for p in pairs(state.collide) do pcall(function() p.CanCollide = true end) end
    table.clear(state.collide)
end

local function startAmmoLoop()
    if state.ammoThread then return end
    state.ammoThread = task.spawn(function()
        while state.active do
            if not isShootingRange() and AXIS.RageMode == "Teleport" then
                -- Teleport modunda sabit aralikli atis (Orbit/Void kendi icinde ates eder)
                if not handleAmmo() and shouldShoot() then
                    local tc = state.target and state.target.Character
                    local head = tc and (tc:FindFirstChild("Head") or getRoot(tc))
                    if head then doFire(head) end
                end
            end
            task.wait(math.max(0.01, clampNum(AXIS.RageTeleportDelay, 0.01, 1, 0.04)))
        end
        state.ammoThread = nil
    end)
end

local function startModeLoops(myGen)
    if AXIS.RageMode == "Teleport" then return end   -- gercek teleport: csync yok
    if AXIS.RageMode == "Void" then startVoidLoop(myGen)
    elseif AXIS.RageMode == "Orbit" then enableVoidCsync()
    else startCsync() end
end

-- VOID SPAM ("meowlua private" start_void_spam uyarlamasi)
-- Her Heartbeat: Humanoid Physics durumuna alinir, kok parca N kez (Iterations) son konumdan
-- rastgele isaretli Min..Max mesafeye atilir (-932098, 834783 gibi), hiz sifirlanir.
local lastVsPos = nil
local function meowVoid(root)
    local hum = lp.Character and lp.Character:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return end
    if hum:GetState() ~= Enum.HumanoidStateType.Physics then pcall(function() hum:ChangeState(Enum.HumanoidStateType.Physics) end) end
    local iters = math.floor(clampNum(AXIS.RageVoidIterations, 1, 500, 500))
    local minD = clampNum(AXIS.RageVoidMinDist, 1, 500000, 1)
    local maxD = math.max(clampNum(AXIS.RageVoidRange, 1, 500000, 50000), minD)
    for _ = 1, iters do
        local base = lastVsPos or root.Position
        local d = minD + math.random() * (maxD - minD)
        local sx = math.random() < 0.5 and 1 or -1
        local sy = math.random() < 0.5 and 1 or -1
        local sz = math.random() < 0.5 and 1 or -1
        local p = Vector3.new(base.X + sx * d * math.random(), base.Y + sy * d * math.random(), base.Z + sz * d * math.random())
        -- koordinatlar cok buyurse basa sar (float hassasiyeti bozulmasin)
        if math.abs(p.X) > 5e6 or math.abs(p.Y) > 5e6 or math.abs(p.Z) > 5e6 then
            p = Vector3.new(math.random(-1000000, 1000000), math.random(-1000000, 1000000), math.random(-1000000, 1000000))
        end
        root.CFrame = CFrame.new(p)
        lastVsPos = p
    end
    root.AssemblyLinearVelocity = Vector3.zero
    root.AssemblyAngularVelocity = Vector3.zero
    if GameState ~= "RAGE_VOIDSPAM" then setGameState("RAGE_VOIDSPAM") end
end

local function leaveVoid()
    if state.voiding then
        state.voiding = false
        lastVsPos = nil
        local hum = lp.Character and lp.Character:FindFirstChildOfClass("Humanoid")
        if hum and hum:GetState() == Enum.HumanoidStateType.Physics and not AXIS.RageHover then
            pcall(function() hum:ChangeState(Enum.HumanoidStateType.GettingUp) end)
        end
    end
end

local function stepRagebot(dt, myGen)
    if not state.active then return end
    AXIS.RageMode = "Teleport"   -- sadece klasik teleport
    if isShootingRange() then
        if not state.suspended then
            state.suspended = true
            state.target, state.randPos, state.voidTargetCF, state.voidExposed = nil, nil, nil, false
            stopCsync() disableVoidCsync()
        end
        return
    end
    if state.suspended then state.suspended = false startModeLoops(myGen) end
    -- mod degistiyse donguleri yeniden kur
    if lastMode ~= AXIS.RageMode then
        lastMode = AXIS.RageMode
        stopCsync() disableVoidCsync()
        rbGen += 1
        myGen = rbGen
        state.gen = myGen
        startModeLoops(myGen)
    end
    noclipTick()

    local root = getRoot(lp.Character)
    if not root then return end
    if not inMatch() then
        clearCsync()
        state.target = nil
        if AXIS.RageMode == "Orbit" then enterVoidState() else setStatus() end
        return
    end

    local now = tick()
    if state.target and playerIsDead(state.target) then state.target = nil end
    if state.target and isInvincible(state.target) then state.target = nil end   -- korumaliysa baska hedefe gec
    if now - rbTgtT >= 0.05 or not state.target then
        rbTgtT = now
        local t = getBestTarget()
        if t then state.target = t end
    end
    if not state.target then
        clearCsync()
        if AXIS.RageMode == "Teleport" and AXIS.RageVoidSpamAfterKill and not anyEnemyAlive() then
            state.voiding = true       -- TUM dusmanlar oldu -> void spam (Heartbeat'te)
        elseif AXIS.RageMode == "Orbit" then
            enterVoidState()
        else
            leaveVoid()            -- hover acikken son noktada havada beklenir
            setStatus()
        end
        return
    end

    local tc = state.target.Character
    local tr = getRoot(tc)
    local head = tc and (tc:FindFirstChild("Head") or tr)
    if not tr or not head then state.target = nil setStatus() return end
    if isNearOtherMatch(tr.Position, state.target) then
        state.target, state.randPos = nil, nil
        if AXIS.RageMode == "Orbit" or AXIS.RageMode == "Void" then enterVoidState() else clearCsync() setStatus() end
        return
    end
    setStatus()
    autoEquipPreferred()

    local mode = AXIS.RageMode
    if mode == "Void" then return end
    if mode == "Orbit" and (now < (state.hideOrbitUntil or 0) or handleAmmo()) then
        state.voidTargetCF, state.voidExposed = nil, false
        enterVoidState()
        return
    end

    -- MERMI DEGISIRKEN VOID SPAM: reload / silah degisimi / equip suresince void'de kal.
    -- Shotgun kullaniyorsan mermi degisimi baslayinca en az "Shotgun Void Time" (vars. 5 sn) void'de bekle.
    if mode == "Teleport" then
        local busy = handleAmmo() or isEquipping()
        if busy then
            if not state.wasBusy then
                state.wasBusy = true
                if isUsingShotgun() then
                    state.reloadVoidUntil = math.max(state.reloadVoidUntil or 0, now + clampNum(AXIS.RageShotgunVoidTime, 5, 30, 5))
                end
            end
            state.reloadVoidUntil = math.max(state.reloadVoidUntil or 0, now + 0.05)
        else
            state.wasBusy = false
        end
        if AXIS.RageVoidOnReload and now < (state.reloadVoidUntil or 0) then
            state.voiding = true
            return
        end
    end
    leaveVoid()

    local targetPos
    if mode == "Teleport" then
        -- KLASIK TELEPORT: hedefin tam ustune, secilen stud kadar (-5..50)
        targetPos = tr.Position + Vector3.new(0, stud(), 0)
    elseif isRiotShield(state.target) then
        targetPos = shieldPos(tr)
    elseif mode == "Underground" then
        targetPos = head.Position - tr.CFrame.LookVector * dist() + Vector3.new(0, stud(), 0)
    elseif mode == "Teleport" then
        if AXIS.RageRandomMovement then
            if not state.randPos or (now - (state.randT or 0)) >= 0.08 then
                state.randT = now
                local a = math.random() * math.pi * 2
                state.randPos = pickOffset(tr, head) + Vector3.new(math.cos(a), 0, math.sin(a)) * (math.random() * 1.05)
            end
            targetPos = state.randPos
        else
            targetPos = pickOffset(tr, head)
        end
    else -- Orbit
        orbitAngle += dt * math.max(1, clampNum(AXIS.RageStrafeSpeed, 1, 20, 5) * 1.5)
        local r = math.max(dist(), 1.25)
        targetPos = head.Position + Vector3.new(math.cos(orbitAngle) * r, stud(), math.sin(orbitAngle) * r)
    end

    if isNearOtherMatch(targetPos, state.target) then
        state.randPos = nil
        if mode == "Orbit" then enterVoidState() else clearCsync() setStatus() end
        return
    end

    -- stud ~ kafa yuksekligi ise bakis vektoru sifir olur (NaN CFrame) -> duz CFrame kullan
    local faceCF = ((head.Position - targetPos).Magnitude > 0.1) and CFrame.new(targetPos, head.Position) or CFrame.new(targetPos)
    if AXIS.RageAntiAim then
        aaPhase += dt * 20
        faceCF = faceCF * CFrame.Angles(0, math.rad(math.sin(aaPhase) * 70), 0)
    end

    if mode == "Orbit" then
        state.voidExposed, state.voidTargetCF = true, faceCF
        setCsync(faceCF, targetPos, dt)
        setStatus()
        if shouldShoot() then doFire(head) end     -- hyper: her karede rapid fire
    elseif mode == "Teleport" then
        -- GERCEK TELEPORT: karakter kendi ekraninda da dusmanin ustune isinlanir
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
        root.CFrame = faceCF
        state.lastTpPos = targetPos
        state.holdCF = faceCF          -- havada sabit kalma noktasi
    else
        setCsync(faceCF, targetPos, dt)
        if mode == "Underground" then
            if shouldShoot() and not handleAmmo() then doFire(head) end
        elseif now >= (state.nextTeleportAt or 0) then
            state.nextTeleportAt = now + math.max(0.01, clampNum(AXIS.RageTeleportDelay, 0.01, 1, 0.04))
        end
    end
end

local function startRagebot()
    if state.active then return end
    local r0 = getRoot(lp.Character)
    state.home = r0 and r0.CFrame or nil          -- kapatinca geri donulecek yer
    state.active, state.suspended = true, isShootingRange()
    rbGen += 1
    local myGen = rbGen
    state.gen = myGen
    lastMode = AXIS.RageMode
    startAmmoLoop()
    if not state.suspended then startModeLoops(myGen) end
    state.voidSpamConn = RunService.Heartbeat:Connect(function()
        -- HAVADA SABIT KAL: void'de degilken son teleport noktasinda tut (yercekimiyle dusmez)
        if state.active and not state.voiding and state.holdCF and AXIS.RageHover then
            local r = getRoot(lp.Character)
            local hum = lp.Character and lp.Character:FindFirstChildOfClass("Humanoid")
            if r and hum and hum.Health > 0 then
                if not hum.PlatformStand then hum.PlatformStand = true end
                r.AssemblyLinearVelocity = Vector3.zero
                r.AssemblyAngularVelocity = Vector3.zero
                r.CFrame = state.holdCF
            end
        end
        if state.active and state.voiding then
            local r = getRoot(lp.Character)
            if r then
                local ok, err = pcall(meowVoid, r)
                if not ok and AXIS.ShowDebug then warn("[axis] void spam: " .. tostring(err)) end
            end
        end
    end)
    state.conn = RunService.Stepped:Connect(function(_, dt)
        local ok, err = pcall(stepRagebot, dt, state.gen)
        if not ok then stats.Errors += 1 if AXIS.ShowDebug then warn("[axis] ragebot: " .. tostring(err)) end end
    end)
    state.charConn = lp.CharacterAdded:Connect(function(char)
        stopCsync() disableVoidCsync()
        state.target, state.voidExposed, state.hideOrbitUntil, state.lastTpPos, state.holdCF = nil, false, 0, nil, nil
        local hrp = char:WaitForChild("HumanoidRootPart", 10)
        if hrp then state.home = hrp.CFrame end
        task.wait(0.5)
        if state.active then startModeLoops(state.gen) end
    end)
    log("Ragebot (Harion) acildi: " .. tostring(AXIS.RageMode))
end

local function stopRagebot()
    if not state.active then return end
    rbGen += 1
    state.active = false
    if state.conn then state.conn:Disconnect() state.conn = nil end
    if state.charConn then state.charConn:Disconnect() state.charConn = nil end
    if state.voidSpamConn then state.voidSpamConn:Disconnect() state.voidSpamConn = nil end
    state.target, state.voidExposed, state.nextTeleportAt, state.ammoActionAt, state.hideOrbitUntil = nil, false, 0, 0, 0
    state.randPos, state.randT, state.lastFakePos = nil, 0, nil
    rbInMatchT, rbInMatch = 0, false
    stopCsync()
    disableVoidCsync()
    restoreCollide()
    -- kapatinca aninda eski yerine don
    local r1 = getRoot(lp.Character)
    if r1 and state.home then
        r1.AssemblyLinearVelocity = Vector3.zero
        r1.AssemblyAngularVelocity = Vector3.zero
        r1.CFrame = state.home
    end
    state.home, state.lastTpPos = nil, nil
    state.voiding, state.wasBusy, state.reloadVoidUntil, state.holdCF = false, false, 0, nil
    local hm0 = lp.Character and lp.Character:FindFirstChildOfClass("Humanoid")
    if hm0 then pcall(function() hm0.PlatformStand = false end) end
    lastVsPos = nil
    local hm = lp.Character and lp.Character:FindFirstChildOfClass("Humanoid")
    if hm and hm:GetState() == Enum.HumanoidStateType.Physics then pcall(function() hm:ChangeState(Enum.HumanoidStateType.GettingUp) end) end
    setGameState("IDLE")
    log("Ragebot kapandi (csync: zaten eski yerindesin)")
end

-- HARION WEAPONS: FAST SHOOT ("Harion enhancement.lua" applyInfoOptions uyarlamasi)
-- Yerel esyalarin Info'sunda cooldown'lar 0, recoil/spread 0, mermi hizi max; kapaninca geri alinir.
-- ============================================================
;(function()
local cache = setmetatable({}, { __mode = "k" })
local KEYS = { "ShootCooldown", "QuickShotCooldown", "SpinCooldown", "DashCooldown", "Cooldown",
    "BuildCooldown", "AttackCooldown", "HeavyAttackCooldown", "BurstCooldown" }

local function remember(info, key)
    if info[key] == nil then return nil end
    local c = cache[info]
    if not c then c = {} cache[info] = c end
    if c[key] == nil then c[key] = info[key] end
    return c[key]
end

local function apply(info, fast)
    local recoil = remember(info, "ShootRecoil")
    if recoil ~= nil then info.ShootRecoil = fast and 0 or recoil end
    local spread = remember(info, "ShootSpread")
    if spread ~= nil then info.ShootSpread = fast and 0 or spread end
    local speed = remember(info, "ProjectileSpeed")
    if speed ~= nil then info.ProjectileSpeed = fast and 99999999 or speed end
    for _, key in ipairs(KEYS) do
        local original = remember(info, key)
        if original ~= nil then info[key] = fast and 0 or original end
    end
end

local applied = false
loop(0.2, function()
    if AXIS.HarionFastShoot then
        local f = localFighter()
        local items = type(f) == "table" and rawget(f, "Items") or nil
        if type(items) == "table" then
            for _, it in pairs(items) do
                local info = type(it) == "table" and it.Info or nil
                if type(info) == "table" then pcall(apply, info, true) end
            end
        end
        applied = true
    elseif applied then
        for info in pairs(cache) do pcall(apply, info, false) end
        applied = false
    end
end)
table.insert(CleanupFns, function() for info in pairs(cache) do pcall(apply, info, false) end end)
end)() -- HARION WEAPONS

        local cachePath = "AXIS/lib/" .. file:gsub("/", "_")
        local code
        if type(isfile) == "function" and type(readfile) == "function" then
            pcall(function() if isfile(cachePath) then code = readfile(cachePath) end end)
        end

                code = body

                        if type(isfolder) == "function" and type(makefolder) == "function" then
                            if not isfolder("AXIS") then makefolder("AXIS") end

    -- execute edildiginde tum toggle'lar kapali baslar, kullanici kendisi acar.
    -- (Matchmaking teleportundan gelindiyse acik ozellikler oldugu gibi kalir.)
    if not Persist.resuming then
        for id, toggle in pairs(Toggles) do
            if type(id) == "string" and id:sub(1, 5) == "AXIS_" and toggle.Value == true and not ALWAYS_ON[id:sub(6)] then
                pcall(function() toggle:SetValue(false) end)
            end
        end
        for k, v in pairs(AXIS) do
            if type(v) == "boolean" and not ALWAYS_ON[k] then AXIS[k] = false end
        end
    end

    Library:SetWatermarkVisibility(true)
    loop(0.5, function()
        if Library.Unloaded then return end
        Library:SetWatermark(string.format("AXIS v28 | %s | Shots %d | TP %d | Err %d",
            GameState, stats.Shots, stats.Teleports, stats.Errors))
    end)

    Library:OnUnload(function()
        Library.Unloaded = true
        if genv.AXIS_CLEANUP then pcall(genv.AXIS_CLEANUP) end
    end)

    Library:Notify("RIVALS RAGEBOT v28 yuklendi (RightShift: menu)", 5)
end

-- temizlik UI'yi de kapatsin (tekrar calistirmada eski menu kalmasin)
do
    local baseCleanup = genv.AXIS_CLEANUP
    local cleaning = false
    genv.AXIS_CLEANUP = function()
        if cleaning then return end
        cleaning = true
        pcall(baseCleanup)
        if UI_Library and not UI_Library.Unloaded then pcall(function() UI_Library:Unload() end) end
    end
end

-- Auto Execute: yuklemede bir kez kuyruga al (her teleportta yeni sunucuda calisir)
if AXIS.AutoExecuteOnTeleport then task.defer(Persist.queue) end

-- Ayarlari duzenli kaydet (teleport olayi hic gelmese bile dosya guncel kalir)
loop(2, function()
    if not Persist.teleporting and not Persist.pending then Persist.save(false) end
end)

-- Teleport baslarken/surerken: isaretli kaydet (autoexec ile calisan kopya da devam modunu anlar)
track(lp.OnTeleport:Connect(function(state)
    if state == Enum.TeleportState.Started or state == Enum.TeleportState.InProgress then
        Persist.teleporting = true
        Persist.save(true)
        if AXIS.AutoExecuteOnTeleport then Persist.queue() end
    elseif state == Enum.TeleportState.Failed then
        Persist.teleporting = false
        Persist.save(false)
    end
end))

-- Teleport sonrasi: acik ozellikleri ancak karakter + fighter hazir olunca geri ac
-- (yukleme sirasinda ragebot/void/hook'larin hemen baslamasi donmaya yol aciyordu)
if Persist.resuming and Persist.pending then
    task.spawn(function()
        local t0 = tick()
        while Running and tick() - t0 < 30 do
            local char = lp.Character
            if char and char:FindFirstChild("HumanoidRootPart") and localFighter() then break end
            task.wait(0.25)
        end
        task.wait(1)
        if not Running then return end
        local toggles = genv.Toggles
        for k in pairs(Persist.pending) do
            local tg = toggles and toggles["AXIS_" .. k]
            if tg then pcall(function() tg:SetValue(true) end) else AXIS[k] = true end
        end
        Persist.pending = nil
        log("Teleport sonrasi ozellikler geri acildi")
    end)
end

    local t0 = tick()
    while Running and not lp.Character and tick() - t0 < 20 do task.wait(0.25) end
    task.wait(1)
    if not Running then return end
    local ok, err = pcall(buildUI)
    if not ok then warn("[axis] UI hatasi: " .. tostring(err)) end
end)
